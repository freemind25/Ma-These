    72→function calculateBalanceScore(chapters: ThesisAnalysis[]): number {
    73→  if (chapters.length === 0) return 100;
    74→  const wordCounts = chapters.map((c) => c.wordCount);
    75→  const total = wordCounts.reduce((a, b) => a + b, 0);
    76→  if (total === 0) return 100;
    77→  const average = total / wordCounts.length;
    78→  const variance = wordCounts.reduce((sum, wc) => sum + Math.pow(wc - average, 2), 0) / wordCounts.length;
    79→  const stdDev = Math.sqrt(variance);
    80→  // 100% = perfectly balanced (stdDev = 0), lower = more imbalance
    81→  // Using coefficient of variation: cv = stdDev / mean
    82→  const cv = average > 0 ? stdDev / average : 0;
    83→  return Math.max(0, Math.min(100, Math.round((1 - cv) * 100)));
    84→}
    85→
    86→function getBalanceColor(score: number): string {
    87→  if (score >= 80) return "text-emerald-600 dark:text-emerald-400";
    88→  if (score >= 60) return "text-amber-600 dark:text-amber-400";
    89→  return "text-red-500 dark:text-red-400";
    90→}
    91→
    92→function getBalanceBgColor(score: number): string {
    93→  if (score >= 80) return "bg-emerald-500";
    94→  if (score >= 60) return "bg-amber-500";
    95→  return "bg-red-500";
    96→}
    97→
    98→function getBalanceLabel(score: number): string {
    99→  if (score >= 90) return "Excellent";
   100→  if (score >= 80) return "Bon";
   101→  if (score >= 60) return "Moyen";
   102→  if (score >= 40) return "Déséquilibré";
   103→  return "Très déséquilibré";
   104→}
   105→
   106→function getDiffIcon(diff: number, average: number) {
   107→  if (average === 0) return <Minus className="h-3.5 w-3.5 text-muted-foreground" />;
   108→  const pct = Math.abs(diff / average) * 100;
   109→  if (pct <= 10) return <CheckCircle2 className="h-3.5 w-3.5 text-emerald-600" />;
   110→  if (diff > 0) return <TrendingUp className="h-3.5 w-3.5 text-amber-600" />;
   111→  return <TrendingDown className="h-3.5 w-3.5 text-red-500" />;
   112→}
