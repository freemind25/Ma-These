   321→
   322→  // Quality assessment state

   325→
   326→  // Data extraction state
   327→  const [extractionData, setExtractionData] = useState<Record<string, string>>({});
   328→  const [selectedExtractionArticle, setSelectedExtractionArticle] = useState<string>("a1");

   330→  // AI assistant state
   331→  const [aiPrompt, setAiPrompt] = useState("");
   332→  const [aiLoading, setAiLoading] = useState(false);
   333→  const [aiResult, setAiResult] = useState("");
   334→
   335→  // Dialog states
   336→  const [addCriteriaOpen, setAddCriteriaOpen] = useState(false);
   337→  const [newCriteriaText, setNewCriteriaText] = useState("");
   338→  const [newCriteriaType, setNewCriteriaType] = useState<"inclusion" | "exclusion">("inclusion");
   339→  const [addKeywordOpen, setAddKeywordOpen] = useState(false);
   340→  const [newKeyword, setNewKeyword] = useState("");
   341→
   342→  // ─── Helper functions ───
   343→
   344→  const updatePrismaCount = (stageId: string, subStageId: string | null, newCount: number) => {
   345→    setPrismaStages((prev) =>
   346→      prev.map((stage) => {
   347→        if (stage.id === stageId) {
   348→          if (!subStageId) return { ...stage, count: newCount };
   349→          return {
   350→            ...stage,
   351→            count: stage.subStages?.reduce((acc, s) => (s.id === subStageId ? acc + newCount : acc + s.count), 0) ?? newCount,
   352→            subStages: stage.subStages?.map((s) => (s.id === subStageId ? { ...s, count: newCount } : s)),
   353→          };
   354→        }
   355→        return stage;
   356→      })
   357→    );
   358→  };
   359→
   360→  const updateArticleDecision = (articleId: string, decision: ScreeningDecision) => {
   361→    setArticles((prev) =>
   362→      prev.map((a) => {
   363→        if (a.id === articleId) {
   364→          const stageMap: Record<string, ArticleItem["stage"]> = {
   365→            include: "included",
   366→            exclude: "screening",
   367→            maybe: "eligibility",
   368→          };
   369→          return { ...a, decision, stage: decision ? stageMap[decision] : a.stage };
   370→        }
   371→        return a;
   372→      })
   373→    );
   374→  };
   375→
   376→  const updateArticleNotes = (articleId: string, notes: string) => {
   377→    setArticles((prev) => prev.map((a) => (a.id === articleId ? { ...a, notes } : a)));
   378→  };
   379→
   380→  const addCriterion = () => {
   381→    if (!newCriteriaText.trim()) return;
   382→    const newCrit: Criterion = {
   383→      id: `c-${Date.now()}`,
   384→      text: newCriteriaText.trim(),
   385→      type: newCriteriaType,
   386→    };
   387→    setCriteria((prev) => [...prev, newCrit]);
   388→    setNewCriteriaText("");
   389→    setAddCriteriaOpen(false);
   390→    toast.success("Critère ajouté");
   391→  };
   392→
   393→  const removeCriterion = (id: string) => {
   394→    setCriteria((prev) => prev.filter((c) => c.id !== id));
   395→    toast.success("Critère supprimé");
   396→  };
   397→
   398→  const toggleDatabase = (db: string) => {
   399→    setSelectedDatabases((prev) =>
   400→      prev.includes(db) ? prev.filter((d) => d !== db) : [...prev, db]
   401→    );
   402→  };
   403→
   404→  const handleAddKeyword = () => {
   405→    if (!newKeyword.trim()) return;
   406→    setKeywords((prev) => (prev ? `${prev}; ${newKeyword.trim()}` : newKeyword.trim()));
   407→    setNewKeyword("");
   408→    setAddKeywordOpen(false);
   409→    toast.success("Mot-clé ajouté");
   410→  };
   411→
   412→  const updatePicoField = (id: string, value: string) => {
   413→    setPicoFields((prev) => prev.map((f) => (f.id === id ? { ...f, value } : f)));
   414→  };
   415→
   416→  const toggleQualityItem = (id: string) => {
   417→    setQualityItems((prev) =>
   418→      prev.map((item) => (item.id === id ? { ...item, checked: !item.checked } : item))
   419→    );
   420→  };
   421→
   422→  const updateQualityNote = (id: string, note: string) => {
   423→    setQualityItems((prev) =>
   424→      prev.map((item) => (item.id === id ? { ...item, note } : item))
   425→    );
   426→  };
   427→
   428→  const generateResearchQuestion = useCallback(() => {
   429→    const p = picoFields.find((f) => f.id === "p")?.value || "";
   430→    const i = picoFields.find((f) => f.id === "i")?.value || "";
   431→    const c = picoFields.find((f) => f.id === "c")?.value || "";
   432→    const o = picoFields.find((f) => f.id === "o")?.value || "";
   433→    return `Quelle est l'effet de ${i || "[Intervention]"} sur ${o || "[Résultats]"} chez ${p || "[Population]"} comparativement à ${c || "[Comparaison]"} ?`;
   434→  }, [picoFields]);
   435→
   436→  const handleAiAssist = useCallback(async () => {
   437→    if (!aiPrompt.trim()) return;
   438→    setAiLoading(true);
   439→    setAiResult("");

   441→      const res = await fetch("/api/ai-writing", {
   442→        method: "POST",
   443→        headers: { "Content-Type": "application/json" },
   444→        body: JSON.stringify({
   445→          mode: "revue-litterature",
   446→          prompt: aiPrompt,
   447→          context: `Revue systématique en cours. Cadre PICO: ${generateResearchQuestion()}. Critères: ${criteria.map((c) => c.text).join("; ")}. Bases: ${selectedDatabases.join(", ")}.`,
   448→        }),
   449→      });
   450→      if (!res.ok) throw new Error("Erreur API");
   451→      const data = await res.json();
   452→      setAiResult(data.data?.content || "Aucune réponse générée.");
   453→    } catch {
   454→      toast.error("Erreur lors de la génération IA");
   455→      setAiResult("Impossible de générer une réponse. Vérifiez votre connexion.");
   456→    } finally {
   457→      setAiLoading(false);
   458→    }
   459→  }, [aiPrompt, criteria, selectedDatabases, generateResearchQuestion]);
   460→
   461→  // ─── Computed values ───
   462→
   463→  const filteredArticles = articles.filter((a) => {
   464→    const matchDecision = articleFilter === "all" || a.decision === articleFilter;
   465→    const matchSearch =
   466→      !searchQuery ||
   467→      a.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
   468→      a.authors.toLowerCase().includes(searchQuery.toLowerCase());
   469→    return matchDecision && matchSearch;
   470→  });
   471→
   472→  const includeCount = articles.filter((a) => a.decision === "include").length;
   473→  const excludeCount = articles.filter((a) => a.decision === "exclude").length;
   474→  const maybeCount = articles.filter((a) => a.decision === "maybe").length;
   475→  const totalCount = articles.length;
   476→  const qualityChecked = qualityItems.filter((q) => q.checked).length;
   477→  const qualityPercent = Math.round((qualityChecked / qualityItems.length) * 100);
   478→  const researchQuestion = generateResearchQuestion();
   479→
   480→  return (
   481→    <div className="max-w-6xl mx-auto flex flex-col gap-6 p-6">
   482→      {/* ─── Header ─── */}
   483→      <div>
   484→        <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   485→          <GitBranch className="h-6 w-6 text-primary" />
   486→          Outils SLR
   487→        </h1>
   488→        <p className="text-sm text-muted-foreground mt-1">
   489→          Revue systématique de la littérature — Diagramme PRISMA, criblage, extraction et évaluation de la qualité
   490→        </p>
   491→      </div>
   492→
   493→      <Separator />
   494→
   495→      {/* ─── Main Tabs ─── */}
   496→      <Tabs defaultValue="prisma" className="flex flex-col gap-6">
   497→        <TabsList className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 w-full h-auto">
   498→          <TabsTrigger value="prisma" className="text-xs py-2">
   499→            <GitBranch className="h-3.5 w-3.5 mr-1.5" />
   500→            PRISMA
   501→          </TabsTrigger>
   502→          <TabsTrigger value="protocol" className="text-xs py-2">
   503→            <FileSpreadsheet className="h-3.5 w-3.5 mr-1.5" />
   504→            Protocole
   505→          </TabsTrigger>
   506→          <TabsTrigger value="screening" className="text-xs py-2">
   507→            <Filter className="h-3.5 w-3.5 mr-1.5" />
   508→            Criblage
   509→          </TabsTrigger>
   510→          <TabsTrigger value="extraction" className="text-xs py-2">
   511→            <ClipboardCheck className="h-3.5 w-3.5 mr-1.5" />
   512→            Extraction
   513→          </TabsTrigger>
   514→          <TabsTrigger value="quality" className="text-xs py-2">
   515→            <CheckCircle2 className="h-3.5 w-3.5 mr-1.5" />
   516→            Qualité
