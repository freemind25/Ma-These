   135→function formatAuthorsAPA(authors: string): string {
   136→  if (!authors.trim()) return "";
   137→  const parts = authors.split(";").map((a) => a.trim()).filter(Boolean);
   138→  if (parts.length === 0) return "";
   139→
   140→  const formatted = parts.map((author) => {
   141→    const tokens = author.split(",").map((t) => t.trim());
   142→    if (tokens.length >= 2) {
   143→      const lastName = tokens[0];
   144→      const initials = tokens
   145→        .slice(1)
   146→        .join("")
   147→        .split(/\s+/)
   148→        .filter(Boolean)
   149→        .map((n) => n.charAt(0).toUpperCase() + ".")
   150→        .join(" ");
   151→      return `${lastName}, ${initials}`;
   152→    }
   153→    // Try to parse "First Last" format
   154→    const words = author.split(/\s+/).filter(Boolean);
   155→    if (words.length >= 2) {
   156→      const lastName = words[words.length - 1];
   157→      const initials = words
   158→        .slice(0, -1)
   159→        .map((n) => n.charAt(0).toUpperCase() + ".")
   160→        .join(" ");
   161→      return `${lastName}, ${initials}`;
   162→    }
   163→    return author;
   164→  });
   165→
   166→  if (formatted.length === 1) return formatted[0];
   167→  if (formatted.length === 2) return `${formatted[0]}, & ${formatted[1]}`;
   168→  return `${formatted.slice(0, -1).join(", ")}, & ${formatted[formatted.length - 1]}`;
   169→}

   171→function generateCitationAPA(ref: ReferenceData): string {
   172→  const authorsFormatted = formatAuthorsAPA(ref.authors);
   173→  if (!authorsFormatted || !ref.title) return "";
   174→
   175→  const year = ref.year ? `(${ref.year}).` : "(s.d.).";
   176→
   177→  switch (ref.type) {
   178→    case "journal": {
   179→      let citation = `${authorsFormatted} ${year} ${ref.title}.`;
   180→      if (ref.journal) {
   181→        citation += ` *${ref.journal}*`;
   182→        if (ref.volume) {
   183→          citation += `, *${ref.volume}*`;
   184→          if (ref.issue) citation += `(${ref.issue})`;
   185→        }
   186→        if (ref.pages) citation += `, ${ref.pages}`;
   187→        citation += ".";
   188→      }
   189→      if (ref.doi) {
   190→        citation += ` https://doi.org/${ref.doi.replace(/^https?:\/\/doi\.org\//, "")}`;
   191→      }
   192→      return citation;
   193→    }
   194→    case "book": {
   195→      let citation = `${authorsFormatted} ${year} *${ref.title}*`;
   196→      if (ref.edition) citation += ` (${ref.edition} éd.)`;
   197→      citation += ".";
   198→      if (ref.publisher) {
   199→        const parts = [ref.city, ref.publisher].filter(Boolean);
   200→        if (parts.length > 0) citation += ` ${parts.join(": ")}.`;
   201→      }
   202→      if (ref.doi) {
   203→        citation += ` https://doi.org/${ref.doi.replace(/^https?:\/\/doi\.org\//, "")}`;
   204→      }
   205→      return citation;
   206→    }
   207→    case "chapter": {
   208→      let citation = `${authorsFormatted} ${year} ${ref.title}.`;
   209→      if (ref.editor) {
   210→        const editorFormatted = formatAuthorsAPA(ref.editor);
   211→        if (editorFormatted) {
   212→          citation += ` Dans ${editorFormatted} (Éd.), *${ref.bookTitle || ""}*`;
   213→          if (ref.edition) citation += ` (${ref.edition} éd.)`;
   214→          citation += ".";
   215→        }
   216→      } else if (ref.bookTitle) {
   217→        citation += ` Dans *${ref.bookTitle}*.`;
   218→      }
   219→      if (ref.pages) citation += ` (pp. ${ref.pages}).`;
   220→      if (ref.publisher) {
   221→        const parts = [ref.city, ref.publisher].filter(Boolean);
   222→        if (parts.length > 0) citation += ` ${parts.join(": ")}.`;
   223→      }
   224→      if (ref.doi) {
   225→        citation += ` https://doi.org/${ref.doi.replace(/^https?:\/\/doi\.org\//, "")}`;
   226→      }
   227→      return citation;
   228→    }
   229→    case "website": {
   230→      let citation = `${authorsFormatted} ${year} ${ref.title}.`;
   231→      if (ref.siteName) citation += ` ${ref.siteName}.`;
   232→      if (ref.url) citation += ` ${ref.url}`;
   233→      return citation;
   234→    }
   235→    case "dissertation": {
   236→      let citation = `${authorsFormatted} ${year} *${ref.title}*`;
   237→      if (ref.publisher || ref.university) {
   238→        citation += ` [Thèse de doctorat, ${ref.university || ref.publisher}].`;
   239→      }
   240→      if (ref.doi) {
   241→        citation += ` https://doi.org/${ref.doi.replace(/^https?:\/\/doi\.org\//, "")}`;
   242→      }
   243→      return citation;
   244→    }
   245→    default:
   246→      return "";
   247→  }
   248→}

   250→function generateInTextCitation(ref: ReferenceData, parenthetical: boolean = true): string {
   251→  const authorsFormatted = formatAuthorsAPA(ref.authors);
   252→  if (!authorsFormatted) return "";
   253→
   254→  // Get last name only for in-text
   255→  const lastName = authorsFormatted.split(",")[0].trim();
   256→  const year = ref.year || "s.d.";
   257→
   258→  if (parenthetical) {
   259→    return `(${lastName}, ${year})`;
   260→  }
   261→  return `${lastName} (${year})`;
   262→}

   264→function generateBibTeX(ref: ReferenceData): string {
   265→  const typeMap: Record<ReferenceType, string> = {
   266→    journal: "article",
   267→    book: "book",
   268→    chapter: "incollection",
   269→    website: "misc",
   270→    dissertation: "phdthesis",
   271→  };
   272→
   273→  const citeKey = ref.authors
   274→    .split(",")[0]
   275→    .trim()
   276→    .replace(/[^a-zA-Z]/g, "")
   277→    .toLowerCase();
   278→  const citeKeyClean = `${citeKey}${ref.year || ""}`;
   279→
   280→  const fields: string[] = [];
   281→  fields.push(`  author = {${ref.authors}}`);
   282→  if (ref.year) fields.push(`  year = {${ref.year}}`);
   283→  fields.push(`  title = {${ref.title}}`);
   284→
   285→  switch (ref.type) {
   286→    case "journal":
   287→      if (ref.journal) fields.push(`  journal = {${ref.journal}}`);
   288→      if (ref.volume) fields.push(`  volume = {${ref.volume}}`);
   289→      if (ref.issue) fields.push(`  number = {${ref.issue}}`);
   290→      if (ref.pages) fields.push(`  pages = {${ref.pages}}`);
   291→      break;
   292→    case "book":
   293→      if (ref.publisher) fields.push(`  publisher = {${ref.publisher}}`);
   294→      if (ref.city) fields.push(`  address = {${ref.city}}`);
   295→      if (ref.edition) fields.push(`  edition = {${ref.edition}}`);
   296→      break;
   297→    case "chapter":
   298→      if (ref.bookTitle) fields.push(`  booktitle = {${ref.bookTitle}}`);
   299→      if (ref.editor) fields.push(`  editor = {${ref.editor}}`);
   300→      if (ref.publisher) fields.push(`  publisher = {${ref.publisher}}`);
   301→      if (ref.pages) fields.push(`  pages = {${ref.pages}}`);
   302→      break;
   303→    case "website":
   304→      if (ref.url) fields.push(`  url = {${ref.url}}`);
   305→      if (ref.siteName) fields.push(`  howpublished = {${ref.siteName}}`);
   306→      break;
   307→    case "dissertation":
   308→      if (ref.university) fields.push(`  school = {${ref.university}}`);
   309→      if (ref.publisher) fields.push(`  school = {${ref.publisher}}`);
   310→      break;
   311→  }
   312→
   313→  if (ref.doi) fields.push(`  doi = {${ref.doi}}`);
   314→  if (ref.url && ref.type !== "website") fields.push(`  url = {${ref.url}}`);
   315→
   316→  return `@${typeMap[ref.type]}{${citeKeyClean},\n${fields.join(",\n")}\n}`;
   317→}

   320→// BibTeX Parser (simple)
   321→// ═══════════════════════════════════════════════
   322→
   323→function parseBibTeXEntries(raw: string): ReferenceData[] {
   324→  const entries: ReferenceData[] = [];
   325→  const entryRegex = /@(\w+)\{([^,]+),\s*([\s\S]*?)\n\}/g;
   326→  let match: RegExpExecArray | null;
   327→
   328→  while ((match = entryRegex.exec(raw)) !== null) {
   329→    const type = match[1].toLowerCase();
   330→    const _entryKey = match[2].trim();
   331→    const body = match[3];
   332→    void _entryKey;
   333→
   334→    const fieldRegex = /\s*(\w+)\s*=\s*\{([^}]*)\}/g;
   335→    const fields: Record<string, string> = {};
   336→    let fieldMatch: RegExpExecArray | null;
   337→
   338→    while ((fieldMatch = fieldRegex.exec(body)) !== null) {
   339→      fields[fieldMatch[1].toLowerCase()] = fieldMatch[2].trim();
   340→    }
   341→
   342→    const ref: ReferenceData = {
   343→      id: crypto.randomUUID(),
   344→      type: "journal",
   345→      authors: fields.author || "",
   346→      year: fields.year || "",
   347→      title: fields.title || "",
   348→      journal: fields.journal || "",
   349→      volume: fields.volume || "",
   350→      issue: fields.number || "",
   351→      pages: fields.pages || "",
   352→      doi: fields.doi || "",
   353→      publisher: fields.publisher || "",
   354→      city: fields.address || "",
   355→      edition: fields.edition || "",
   356→      editor: fields.editor || "",
   357→      bookTitle: fields.booktitle || "",
   358→      siteName: fields.howpublished || "",
   359→      url: fields.url || "",
   360→      university: fields.school || "",
   361→    };
   362→
   363→    if (type === "book") ref.type = "book";
   364→    else if (type === "incollection") ref.type = "chapter";
   365→    else if (type === "phdthesis" || type === "mastersthesis") ref.type = "dissertation";
   366→    else if (type === "misc" || type === "online") ref.type = "website";
   367→    else ref.type = "journal";
   368→
   369→    if (ref.authors || ref.title) entries.push(ref);
   370→  }
   371→
   372→  return entries;
   373→}
