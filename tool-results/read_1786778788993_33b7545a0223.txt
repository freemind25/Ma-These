   477→// HELPERS
   478→// ═══════════════════════════════════════════════════════════════════════
   479→
   480→function getCategoryScore(
   481→  category: CompetencyCategory,
   482→  skills: Record<string, ProficiencyLevel>
   483→): number {
   484→  const catSkills = category.skills;
   485→  if (catSkills.length === 0) return 0;
   486→  const total = catSkills.reduce(
   487→    (sum, s) => sum + (PROFICIENCY_VALUES[skills[s.id] || "Débutant"] || 0),
   488→    0
   489→  );
   490→  return Math.round(total / catSkills.length);
   491→}
   492→
   493→function getOverallScore(skills: Record<string, ProficiencyLevel>): number {
   494→  const allSkills = COMPETENCY_CATEGORIES.flatMap((c) => c.skills);
   495→  if (allSkills.length === 0) return 0;
   496→  const total = allSkills.reduce(
   497→    (sum, s) => sum + (PROFICIENCY_VALUES[skills[s.id] || "Débutant"] || 0),
   498→    0
   499→  );
   500→  return Math.round(total / allSkills.length);
   501→}
   502→
   503→function getScoreLabel(score: number): string {
   504→  if (score >= 90) return "Expert";
   505→  if (score >= 70) return "Avancé";
   506→  if (score >= 45) return "Intermédiaire";
   507→  return "Débutant";
   508→}
   509→
   510→function getScoreColor(score: number): string {
   511→  if (score >= 90) return "text-purple-600 dark:text-purple-400";
   512→  if (score >= 70) return "text-emerald-600 dark:text-emerald-400";
   513→  if (score >= 45) return "text-sky-600 dark:text-sky-400";
   514→  return "text-amber-600 dark:text-amber-400";
   515→}
