    86→function formatFileSize(bytes: number): string {
    87→  if (bytes === 0) return "—";
    88→  const units = ["o", "Ko", "Mo", "Go"];
    89→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
    90→  return `${(bytes / Math.pow(1024, i)).toFixed(1)} ${units[i]}`;
    91→}
    92→
    93→function formatDate(dateStr: string): string {
    94→  try {
    95→    return new Intl.DateTimeFormat("fr-FR", {
    96→      day: "numeric",
    97→      month: "short",
    98→      year: "numeric",
    99→      hour: "2-digit",
   100→      minute: "2-digit",
   101→    }).format(new Date(dateStr));
   102→  } catch {
   103→    return dateStr;
   104→  }
   105→}
   106→
   107→function getFileBadgeColor(mimeType: string): string {
   108→  if (mimeType.includes("pdf")) return "bg-red-500/10 text-red-700 dark:text-red-400";
   109→  if (mimeType.includes("word") || mimeType.includes("document"))
   110→    return "bg-sky-500/10 text-sky-700 dark:text-sky-400";
   111→  if (mimeType.includes("sheet") || mimeType.includes("excel"))
   112→    return "bg-emerald-500/10 text-emerald-700 dark:text-emerald-400";
   113→  if (mimeType.includes("presentation") || mimeType.includes("slides"))
   114→    return "bg-amber-500/10 text-amber-700 dark:text-amber-400";
   115→  if (mimeType.includes("image")) return "bg-purple-500/10 text-purple-700 dark:text-purple-400";
   116→  if (mimeType.includes("text")) return "bg-stone-500/10 text-stone-700 dark:text-stone-400";
   117→  return "bg-muted text-muted-foreground";
   118→}
   119→
   120→function getFileBadgeLabel(mimeType: string): string {
   121→  if (mimeType.includes("pdf")) return "PDF";
   122→  if (mimeType.includes("word") || mimeType.includes("document")) return "DOCX";
   123→  if (mimeType.includes("sheet") || mimeType.includes("excel")) return "XLSX";
   124→  if (mimeType.includes("presentation") || mimeType.includes("slides")) return "PPTX";
   125→  if (mimeType.includes("image")) return "Image";
   126→  if (mimeType.includes("text")) return "Texte";
   127→  return mimeType.split("/").pop()?.toUpperCase() ?? "Fichier";
   128→}

   236→    setCurrentPath((prev) => prev.slice(0, index + 1));

   240→    setCurrentPath((prev) => [...prev, name]);
