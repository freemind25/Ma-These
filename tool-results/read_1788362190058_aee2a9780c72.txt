    70→// ═══════════════════════════════════════════════════
    71→// Config loading/saving via /api/ai-config (httpOnly cookie)
    72→// API keys are NEVER stored in localStorage.
    73→// ═══════════════════════════════════════════════════
    74→
    75→const LEGACY_STORAGE_KEY = "thesisframe-ai-config";
    76→
    77→/** Load non-sensitive config fields from localStorage (provider/model only) */
    78→function loadLocalConfig(): AiProviderConfig {
    79→  if (typeof window === "undefined") return { provider: "zai" };
    80→  try {
    81→    const raw = localStorage.getItem(LEGACY_STORAGE_KEY);
    82→    if (raw) {
    83→      const parsed = JSON.parse(raw) as AiProviderConfig;
    84→      if (parsed.provider) return parsed;
    85→    }
    86→  } catch { /* ignore */ }
    87→  return { provider: "zai" };
    88→}
    89→
    90→/** Save non-sensitive config to localStorage AND full config to server httpOnly cookie */
    91→async function saveConfigToServer(config: AiProviderConfig): Promise<boolean> {
    92→  try {
    93→    const res = await fetch("/api/ai-config", {
    94→      method: "POST",
    95→      headers: { "Content-Type": "application/json" },
    96→      body: JSON.stringify(config),
    97→    });
    98→    if (res.ok) {
    99→      // Store non-sensitive fields in localStorage for UI persistence
   100→      const { apiKey: _, ...safeConfig } = config;
   101→      localStorage.setItem(LEGACY_STORAGE_KEY, JSON.stringify(safeConfig));
   102→      // Notify other hooks
   103→      window.dispatchEvent(new CustomEvent("ai-config-changed"));
   104→      return true;
   105→    }
   106→  } catch { /* ignore */ }
   107→  return false;
   108→}

   125→function AiConfigDialog({
   126→  onOpenChange,

   128→  open: boolean;
   129→  onOpenChange: (v: boolean) => void;

   132→  const [config, setConfig] = useState<AiProviderConfig>(loadLocalConfig);
   133→  const [showKey, setShowKey] = useState(false);
   134→  const [testing, setTesting] = useState(false);
   135→  const [testResult, setTestResult] = useState<"success" | "fail" | null>(null);
   136→  const [dynamicModels, setDynamicModels] = useState<string[]>([]);
   137→  const [loadingModels, setLoadingModels] = useState(false);
   138→  const [hardcodedKeys, setHardcodedKeys] = useState<HardcodedKeyInfo[]>([]);
   139→  const [hasSavedKey, setHasSavedKey] = useState(false);
   140→  const [isDesktop, setIsDesktop] = useState(false);
   141→
   142→  // Set of providers that have a hardcoded key
   143→  const hardcodedSet = useMemo(
   144→    () => new Set(hardcodedKeys.map((k) => k.provider)),
   145→    [hardcodedKeys]
   146→  );
   147→
   148→  // Sync config on mount from localStorage (non-sensitive) + server (hasApiKey + isDesktop)
   149→  useEffect(() => {
   150→    const local = loadLocalConfig();
   151→    setConfig(local);
   152→    fetch("/api/ai-config")
   153→      .then((r) => r.ok ? r.json() : null)
   154→      .then((data) => {
   155→        if (data?.data) {
   156→          setHasSavedKey(!!data.data.hasApiKey);
   157→          setIsDesktop(!!data.data.isDesktop);
   158→          // If desktop and current provider is zai, switch to a usable provider
   159→          if (data.data.isDesktop && local.provider === "zai") {
   160→            setConfig({
   161→              provider: "mistral",
   162→              model: "mistral-small-latest",
   163→            });
   164→          } else {
   165→            setConfig({
   166→              provider: (data.data.provider || local.provider) as AiProviderId,
   167→              model: data.data.model || local.model,
   168→              baseUrl: data.data.baseUrl || local.baseUrl,
   169→            });
   170→          }
   171→        }
   172→      })
   173→      .catch(() => { /* use local config */ });
   174→  }, []);
   175→
   176→  // Fetch hardcoded key info on mount
   177→  useEffect(() => {
   178→    fetch("/api/ai-keys")
   179→      .then((r) => r.json())
   180→      .then((data) => {
   181→        if (Array.isArray(data.providers)) {
   182→          setHardcodedKeys(data.providers);
   183→        }
   184→      })
   185→      .catch(() => { /* ignore */ });

   187→
   188→  // Fetch dynamic models when provider is dynamic and apiKey is set
   189→  const fetchDynamicModels = useCallback(async () => {
   190→    const fields = getProviderFields(config.provider);
   191→    if (!fields.dynamicModels) {
   192→      setDynamicModels([]);
   193→      return;
   194→    }
   195→
   196→    const baseUrl =
   197→      config.baseUrl || PROVIDER_BASE_URLS[config.provider] || "";

   199→    if (!baseUrl || (!config.apiKey && !hardcodedSet.has(config.provider))) {
   200→      setDynamicModels([]);
   201→      return;
   202→    }
   203→
   204→    setLoadingModels(true);
   205→    try {
   206→      const params = new URLSearchParams({ baseUrl });
   207→      if (config.apiKey) params.set("apiKey", config.apiKey);
   208→      params.set("provider", config.provider);
   209→      const res = await fetch(`/api/ai-models?${params}`);
   210→      const data = (await res.json()) as { models?: string[]; error?: string; cached?: boolean };
   211→      if (data.error) {
   212→        toast.error(data.error);
   213→        setDynamicModels([]);
   214→      } else if (data.models && data.models.length > 0) {
   215→        setDynamicModels(data.models);
   216→        // Auto-select first model if none selected or previously selected model not in list
   217→        if (!config.model || !data.models.includes(config.model)) {
   218→          setConfig((prev) => ({ ...prev, model: data.models![0] }));
   219→        }
   220→      } else {
   221→        setDynamicModels([]);
   222→        if (!data.cached) {
   223→          toast.info("Aucun modèle trouvé pour ce fournisseur.");
   224→        }
   225→      }
   226→    } catch (err) {
   227→      toast.error(`Erreur réseau: ${err instanceof Error ? err.message : String(err)}`);
   228→      setDynamicModels([]);
   229→    } finally {
   230→      setLoadingModels(false);
   231→    }
   232→  }, [config.provider, config.baseUrl, config.apiKey, config.model]);
   233→
   234→  // Keep a stable ref to fetchDynamicModels so the effect doesn't re-fire
   235→  // when config.model changes (which fetchDynamicModels itself may set)
   236→  const fetchDynamicModelsRef = useRef(fetchDynamicModels);
   237→  fetchDynamicModelsRef.current = fetchDynamicModels;

   239→  // Auto-fetch when provider is dynamic and apiKey is available (user-provided or hardcoded)
   240→  useEffect(() => {

   242→    if (DYNAMIC_MODEL_PROVIDERS.includes(config.provider) && hasKey) {
   243→      const timer = setTimeout(() => fetchDynamicModelsRef.current(), 300);
   244→      return () => clearTimeout(timer);
   245→    }
   246→    if (!config.apiKey && !hardcodedSet.has(config.provider)) {
   247→      setDynamicModels([]);
   248→    }
   249→  }, [config.provider, config.apiKey, hardcodedSet]);
   250→
   251→  const handleProviderChange = useCallback((value: string) => {
   252→    const newProvider = value as AiProviderId;
   253→    // Auto-fill the default model for this provider
   254→    const defaultModel = PROVIDER_MODELS[newProvider]?.[0];
   255→    // Auto-fill base URL for known providers
   256→    const defaultBaseUrl = PROVIDER_BASE_URLS[newProvider];
   257→
   258→    setConfig((prev) => ({
   259→      provider: newProvider,
   260→      // If the provider has a hardcoded key, leave apiKey empty (server will inject it)
   261→      // Otherwise keep previous key if user already typed one
   262→      apiKey: hardcodedSet.has(newProvider) ? undefined : (newProvider === "zai" ? undefined : prev.apiKey),
   263→      model: defaultModel || prev.model,
   264→      baseUrl: newProvider === "custom" ? prev.baseUrl : (defaultBaseUrl || undefined),
   265→    }));
   266→    setTestResult(null);
   267→    setDynamicModels([]);
   268→  }, [hardcodedSet]);
   269→
   270→  const handleSave = useCallback(async () => {
   271→    const ok = await saveConfigToServer(config);
   272→    if (ok) {
   273→      setAiProvider(config.provider);
   274→      toast.success(`Fournisseur IA : ${getProviderLabel(config.provider)}`);
   275→      onOpenChange(false);
   276→    } else {
   277→      toast.error("Erreur lors de la sauvegarde de la configuration");
   278→    }
   279→  }, [config, setAiProvider, onOpenChange]);
   280→
   281→  const handleTest = useCallback(async () => {
   282→    setTesting(true);
   283→    setTestResult(null);
   284→    try {
   285→      const res = await fetch("/api/ai-test", {
   286→        method: "POST",
   287→        headers: { "Content-Type": "application/json" },
   288→        body: JSON.stringify({
   289→          provider: config.provider,
   290→          apiKey: config.apiKey,
   291→          model: config.model,
   292→          baseUrl: config.baseUrl,
   293→        }),
   294→      });
   295→      const data = (await res.json()) as { ok?: boolean; error?: string };
   296→      if (!res.ok || !data.ok) {
   297→        throw new Error(data.error || `HTTP ${res.status}`);
   298→      }
   299→      setTestResult("success");
   300→      toast.success("Connexion réussie !");
   301→    } catch (err) {
   302→      setTestResult("fail");
   303→      toast.error(
   304→        `Échec : ${err instanceof Error ? err.message : String(err)}`
   305→      );
   306→    } finally {
   307→      setTesting(false);
   308→    }
   309→  }, [config]);
   310→
   311→  const fields = getProviderFields(config.provider);
   312→  const staticModels = PROVIDER_MODELS[config.provider] || [];
   313→  const models = dynamicModels.length > 0 ? dynamicModels : staticModels;
   314→
   315→  // Current provider's hardcoded key info
   316→  const currentHardcoded = hardcodedKeys.find(
   317→    (k) => k.provider === config.provider
   318→  );
   319→
   320→  return (
   321→    <DialogContent className="sm:max-w-lg max-h-[90vh]">
   322→      <DialogHeader>
   323→        <DialogTitle className="flex items-center gap-2">
   324→          <Bot className="h-5 w-5" />
   325→          Fournisseur IA
   326→        </DialogTitle>
   327→        <DialogDescription>
   328→          Choisissez le fournisseur d&apos;intelligence artificielle pour les fonctions IA de Ma Thèse.
   329→        </DialogDescription>
   330→      </DialogHeader>
   331→
   332→      <ScrollArea className="max-h-[65vh] pr-3">
   333→        <div className="flex flex-col gap-4 py-2">
   334→          {/* Provider select with categories */}
   335→          <div className="flex flex-col gap-1.5">
   336→            <Label>Fournisseur</Label>
   337→            <Select value={config.provider} onValueChange={handleProviderChange}>
   338→              <SelectTrigger>
   339→                <SelectValue placeholder="Choisir un fournisseur" />
   340→              </SelectTrigger>
   341→              <SelectContent className="max-h-80">
   342→                {(Object.entries(PROVIDER_CATEGORIES) as [ProviderCategory, typeof PROVIDER_CATEGORIES[ProviderCategory]][]).map(
   343→                  ([catId, cat], idx) => (
   344→                    <SelectGroup key={catId}>
   345→                      {idx > 0 && <SelectSeparator />}
   346→                      <SelectLabel className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">
   347→                        {cat.label}
   348→                        {cat.description && (
   349→                          <span className="font-normal normal-case ml-1.5 text-muted-foreground/70">
   350→                            — {cat.description}
   351→                          </span>
   352→                        )}
   353→                      </SelectLabel>
   354→                      {cat.providers
   355→                        .filter((p) => !(isDesktop && p === "zai"))
   356→                        .map((p) => {
   357→                        const hasHardcoded = hardcodedSet.has(p);
   358→                        const needsKey = providerNeedsKey(p);
   359→                        return (
   360→                          <SelectItem key={p} value={p}>
   361→                            <span className="flex items-center gap-2">
   362→                              {p === "routesme" && <Zap className="h-3.5 w-3.5 text-amber-500" />}
   363→                              {getProviderLabel(p)}
   364→                              {hasHardcoded && (
   365→                                <Badge
   366→                                  variant="secondary"
   367→                                  className="text-[9px] px-1.5 py-0 h-4 bg-emerald-100 dark:bg-emerald-900/40 text-emerald-700 dark:text-emerald-400 border-0"
   368→                                >
   369→                                  <KeyRound className="h-2.5 w-2.5 mr-0.5" />
   370→                                  clé OK
   371→                                </Badge>
   372→                              )}
   373→                              {!hasHardcoded && !needsKey && p === "zai" && (
   374→                                <Badge variant="secondary" className="text-[9px] px-1.5 py-0 h-4">
   375→                                  natif
   376→                                </Badge>
   377→                              )}
   378→                            </span>
   379→                          </SelectItem>
   380→                        );
   381→                      })}
   382→                    </SelectGroup>
   383→                  )
   384→                )}
   385→              </SelectContent>
   386→            </Select>
   387→          </div>
   388→
   389→          {/* Hardcoded key indicator */}
   390→          {currentHardcoded && (
   391→            <div className="rounded-lg border border-emerald-200 dark:border-emerald-800/50 bg-emerald-50 dark:bg-emerald-950/20 p-3 text-[11px] text-emerald-800 dark:text-emerald-300 space-y-1">
   392→              <p className="font-semibold flex items-center gap-1.5">
   393→                <KeyRound className="h-3.5 w-3.5" />
   394→                Clé API pré-configurée
   395→              </p>
   396→              <p className="font-mono text-[10px] opacity-70">
   397→                {currentHardcoded.maskedKey}
   398→              </p>
   399→              <p className="opacity-80">
   400→                Aucune clé à saisir. Le serveur utilisera automatiquement la clé intégrée.
   401→              </p>
   402→            </div>
   403→          )}
   404→
   405→          {/* API Key (only if provider needs key AND has no hardcoded key) */}
   406→          {fields.showApiKey && !currentHardcoded && (
   407→            <div className="flex flex-col gap-1.5">
   408→              <Label>Clé API</Label>
   409→              <div className="relative">
   410→                <Input
   411→                  type={showKey ? "text" : "password"}
   412→                  placeholder={
   413→                    config.provider === "routesme"
   414→                      ? "rm-xxxxxxxxxxxxxxxx"
   415→                      : config.provider === "mistral"
   416→                        ? "xxxxxxxxxxxxxxxxxxxxxxxx"
   417→                        : "sk-..."
   418→                  }
   419→                  value={config.apiKey || ""}
   420→                  onChange={(e) =>
   421→                    setConfig((prev) => ({ ...prev, apiKey: e.target.value }))
   422→                  }
   423→                />
   424→                <Button
   425→                  type="button"
   426→                  variant="ghost"
   427→                  size="icon"
   428→                  className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7"
   429→                  onClick={() => setShowKey(!showKey)}
   430→                >
   431→                  {showKey ? (
   432→                    <EyeOff className="h-3.5 w-3.5" />
   433→                  ) : (
   434→                    <Eye className="h-3.5 w-3.5" />
   435→                  )}
   436→                </Button>
   437→              </div>
   438→              {/* Indicateur: clé déjà enregistrée sur le serveur */}
   439→              {hasSavedKey && !config.apiKey && config.provider !== "zai" && (
   440→                <div className="rounded-md border border-emerald-200 dark:border-emerald-800/50 bg-emerald-50 dark:bg-emerald-950/20 px-3 py-2 text-[11px] text-emerald-700 dark:text-emerald-400 flex items-center gap-1.5">
   441→                  <KeyRound className="h-3 w-3 shrink-0" />
   442→                  <span>Clé API déjà enregistrée. Le test et les fonctions IA utiliseront la clé sauvegardée. Saisissez une nouvelle clé uniquement pour la modifier.</span>

   490→                    />
   491→                  </Button>
   492→                )}
   493→              </div>
   494→              <Select
   495→                value={config.model || ""}
   496→                onValueChange={(v) =>
   497→                  setConfig((prev) => ({ ...prev, model: v }))
   498→                }
   499→              >
   500→                <SelectTrigger>
   501→                  <SelectValue placeholder={
   502→                    loadingModels ? "Chargement des modèles…" : "Choisir un modèle"
   503→                  } />
   504→                </SelectTrigger>
   505→                <SelectContent className="max-h-60 overflow-y-auto">
   506→                  {models.map((m) => (
   507→                    <SelectItem key={m} value={m}>
   508→                      <span className="flex items-center gap-1.5">
   509→                        {m}
   510→                        {m.toLowerCase().includes("free") && (

   523→              {loadingModels && (

   531→
   532→          {/* Model input for dynamic providers when no models loaded */}
   533→          {fields.showModel &&
   534→            fields.dynamicModels &&
   535→            models.length === 0 &&
   536→            !loadingModels && (

   539→                <Input
   540→                  type="text"
   541→                  placeholder="Nom du modèle (ex: GLM5.2-free)"

   544→                    setConfig((prev) => ({ ...prev, model: e.target.value }))
