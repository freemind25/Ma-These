    47→    47→// Circuit Breaker — per-provider state
    48→    48→// ═══════════════════════════════════════════════════════════════
    49→    49→
    50→    50→type CircuitState = "closed" | "open" | "half-open";
    51→    51→
    52→    52→interface ProviderCircuit {
    53→    53→  state: CircuitState;
    54→    54→  failureCount: number;
    55→    55→  lastFailureTime: number; // epoch ms
    56→    56→  successCount: number;  // in half-open, successes needed to close
    57→    57→}
    58→    58→
    59→    59→const CIRCUIT_FAILURE_THRESHOLD = 3;
    60→    60→const CIRCUIT_RESET_TIMEOUT_MS = 30_000; // 30s cooldown before half-open
    61→    61→const CIRCUIT_HALF_OPEN_SUCCESS_NEEDED = 1;
    62→    62→const REQUEST_TIMEOUT_MS = 60_000; // 60s global timeout
    63→    63→
    64→    64→// In-memory circuit state (persists for the lifetime of the server process)
    65→    65→const circuitBreaker = new Map<string, ProviderCircuit>();
    66→    66→
    67→    67→function getCircuit(provider: string): ProviderCircuit {
    68→    68→  let c = circuitBreaker.get(provider);
    69→    69→  if (!c) {
    70→    70→    c = { state: "closed", failureCount: 0, lastFailureTime: 0, successCount: 0 };
    71→    71→    circuitBreaker.set(provider, c);
    72→    72→  }
    73→    73→  return c;
    74→    74→}
    75→    75→
    76→    76→/**
    77→    77→ * Check if a provider's circuit allows requests.
    78→    78→ * If open, check if cooldown has elapsed → transition to half-open.
    79→    79→ */
    80→    80→function canRequest(provider: string): boolean {
    81→    81→  const c = getCircuit(provider);
    82→    82→  if (c.state === "closed") return true;
    83→    83→  if (c.state === "half-open") return true;
    84→    84→  // Open — check cooldown
    85→    85→  if (Date.now() - c.lastFailureTime >= CIRCUIT_RESET_TIMEOUT_MS) {
    86→    86→    c.state = "half-open";
    87→    87→    c.successCount = 0;
    88→    88→    return true;
    89→    89→  }
    90→    90→  return false;
    91→    91→}
    92→    92→
    93→    93→/** Record a successful request */
    94→    94→function recordSuccess(provider: string) {
    95→    95→  const c = getCircuit(provider);
    96→    96→  if (c.state === "half-open") {
    97→    97→    c.successCount++;
    98→    98→    if (c.successCount >= CIRCUIT_HALF_OPEN_SUCCESS_NEEDED) {
    99→    99→      c.state = "closed";
   100→   100→      c.failureCount = 0;
   101→   101→      console.log(`[Circuit] ${provider}: closed (recovered)`);
   102→   102→    }
   103→   103→  } else {
   104→   104→    c.failureCount = 0;
   105→   105→  }
   106→   106→}
   107→   107→
   108→   108→/** Record a failed request */
   109→   109→function recordFailure(provider: string) {
   110→   110→  const c = getCircuit(provider);
   111→   111→  c.failureCount++;
   112→   112→  c.lastFailureTime = Date.now();
   113→   113→  if (c.state === "half-open") {
   114→   114→    // Failed test → back to open
   115→   115→    c.state = "open";
   116→   116→    console.warn(`[Circuit] ${provider}: half-open test failed → open`);
   117→   117→  } else if (c.failureCount >= CIRCUIT_FAILURE_THRESHOLD) {
   118→   118→    c.state = "open";
   119→   119→    console.warn(
   120→   120→      `[Circuit] ${provider}: opened after ${c.failureCount} consecutive failures (cooldown ${CIRCUIT_RESET_TIMEOUT_MS / 1000}s)`
   121→   121→    );
   122→   122→  }
   123→   123→}

   128→   128→export function getCircuitBreakerStatus(): Record<string, { state: CircuitState; failureCount: number; lastFailureTime: number }> {
   129→   129→ const result: Record<string, { state: CircuitState; failureCount: number; lastFailureTime: number }> = {};
   130→   130→  for (const [provider, c] of circuitBreaker) {
   131→   131→ result[provider] = { state: c.state, failureCount: c.failureCount, lastFailureTime: c.lastFailureTime };
   132→   132→  }
   133→   133→  return result;

   229→   229→async function generateWithAPI(
   230→   230→  options: AiCompletionOptions,
   231→   231→  config: AiProviderConfig
   232→   232→): Promise<AiCompletionResult> {
   233→   233→  const baseUrl = getBaseUrl(config.provider, config.baseUrl);
   234→   234→
   235→   235→  if (!baseUrl) {
   236→   236→    throw new Error(
   237→   237→      `Aucune URL de base configurée pour le fournisseur "${config.provider}". ` +
   238→   238→      `Veuillez configurer le fournisseur IA dans Paramètres → Fournisseur IA.`
   239→   239→    );
   240→   240→  }
   241→   241→
   242→   242→  const model = options.model || config.model || "gpt-4o-mini";
   243→   243→
   244→   244→  const url = `${baseUrl}/chat/completions`;
   245→   245→  const headers: Record<string, string> = {
   246→   246→    "Content-Type": "application/json",
   247→   247→  };
   248→   248→
   249→   249→  // Auth: keyless providers skip Authorization header
   250→   250→  if (isKeylessProvider(config.provider)) {
   251→   251→    // No auth needed
   252→   252→  } else if (config.apiKey) {
   253→   253→    if (isAnthropicFormat(config.provider)) {
   254→   254→      headers["x-api-key"] = config.apiKey;
   255→   255→      headers["anthropic-version"] = "2023-06-01";
   256→   256→      headers["anthropic-dangerous-direct-browser-access"] = "true";
   257→   257→    } else {
   258→   258→      headers["Authorization"] = `Bearer ${config.apiKey}`;
   259→   259→    }
   260→   260→  }
   261→   261→
   262→   262→  // Provider-specific extra headers (from freellmapi pattern)
   263→   263→  const extraHeaders = getProviderExtraHeaders(config.provider);
   264→   264→  Object.assign(headers, extraHeaders);
   265→   265→
   266→   266→  const systemMessage = options.messages.find((m) => m.role === "system");
   267→   267→  const nonSystemMessages = options.messages.filter((m) => m.role !== "system");
   268→   268→
   269→   269→  let body: Record<string, unknown>;
   270→   270→  if (isAnthropicFormat(config.provider) && systemMessage) {
   271→   271→    body = {
   272→   272→      model,
   273→   273→      max_tokens: options.maxTokens || 4096,
   274→   274→      temperature: options.temperature ?? 0.7,
   275→   275→      system: systemMessage.content,
   276→   276→      messages: nonSystemMessages.map((m) => ({
   277→   277→        role: m.role,
   278→   278→        content: m.content,
   279→   279→      })),
   280→   280→    };
   281→   281→  } else {
   282→   282→    body = {
   283→   283→      model,
   284→   284→      temperature: options.temperature ?? 0.7,
   285→   285→      max_tokens: options.maxTokens || 4096,
   286→   286→      messages: options.messages.map((m) => ({
   287→   287→        role: m.role,
   288→   288→        content: m.content,
   289→   289→      })),
   290→   290→    };
   291→   291→  }
   292→   292→
   293→   293→  const response = await fetch(url, {
   294→   294→    method: "POST",
   295→   295→    headers,
   296→   296→    body: JSON.stringify(body),
   297→   297→    signal: AbortSignal.timeout(REQUEST_TIMEOUT_MS),
   298→   298→  });
   299→   299→
   300→   300→  if (!response.ok) {
   301→   301→    const err = await buildApiError(response, model);
   302→   302→    throw err;
   303→   303→  }
   304→   304→
   305→   305→  const data = await response.json() as Record<string, unknown>;
   306→   306→  const choices = data.choices as Array<Record<string, unknown>> | undefined;
   307→   307→  const firstChoice = choices?.[0];
   308→   308→  const message = firstChoice?.message as Record<string, unknown> | undefined;
   309→   309→  const content = String(message?.content || "");
   310→   310→
   311→   311→  const usage = data.usage as Record<string, number> | undefined;
   312→   312→
   313→   313→  return {
   314→   314→    content: content.trim(),
   315→   315→    model,
   316→   316→    provider: config.provider,
   317→   317→    routedVia: `${config.provider}/${model}`,
   318→   318→    usage: usage
   319→   319→      ? {
   320→   320→          promptTokens: usage.prompt_tokens || 0,
   321→   321→          completionTokens: usage.completion_tokens || 0,
   322→   322→          totalTokens: usage.total_tokens || 0,
   323→   323→        }
   324→   324→      : undefined,
   325→   325→  };
   326→   326→}

   328→   328→/**
   329→   329→ * Build a structured error from an API response.
   330→   330→ * Returns an error with a `status` property for failover logic.
   331→   331→ */
   332→   332→async function buildApiError(response: Response, model: string): Promise<Error & { status: number; retryable: boolean; authError: boolean }> {
   333→   333→  let errorText = await response.text();
   334→   334→
   335→   335→  try {
   336→   336→    const errJson = JSON.parse(errorText) as {
   337→   337→      error?: { type?: string; message?: string };
   338→   338→      message?: string;
   339→   339→      code?: string;
   340→   340→      detail?: string;
   341→   341→      type?: string;
   342→   342→      model?: string;
   343→   343→    };
   344→   344→    const errType = errJson.error?.type || errJson.type || errJson.code || "";
   345→   345→    const errMsg = errJson.error?.message || errJson.message || errJson.detail || "";
   346→   346→
   347→   347→    if (errType === "all_keys_failed" || response.status === 503) {
   348→   348→      errorText = `Service temporairement indisponible. Modèle "${errJson.model || model}" surchargé. Réessayez dans quelques instants ou changez de modèle.`;
   349→   349→    } else if (errType === "rate_limit_exceeded" || response.status === 429) {
   350→   350→      errorText = `Limite de requêtes atteinte. ${errMsg || "Attendez quelques secondes."}`;
   351→   351→    } else if (
   352→   352→      errType === "invalid_api_key" ||
   353→   353→      errType === "invalid_request_error" ||
   354→   354→      errJson.code === "invalid_api_key" ||
   355→   355→      response.status === 401
   356→   356→    ) {
   357→   357→      errorText = `Clé API invalide. ${errMsg || "Vérifiez votre configuration."}`;
   358→   358→    } else if (response.status === 404) {
   359→   359→      errorText = `Modèle "${model}" introuvable. Vérifiez le nom du modèle.`;
   360→   360→    } else if (errMsg) {
   361→   361→      errorText = errMsg;
   362→   362→    }
   363→   363→  } catch {
   364→   364→    // keep raw errorText
   365→   365→  }
   366→   366→
   367→   367→  const error = new Error(`Erreur IA (${response.status}): ${errorText.slice(0, 300)}`) as Error & {
   368→   368→    status: number;
   369→   369→    retryable: boolean;
   370→   370→    authError: boolean;
   371→   371→  };
   372→   372→  error.status = response.status;
   373→   373→  error.retryable = isRetryableError(response.status);
   374→   374→  error.authError = isAuthError(response.status);
   375→   375→  return error;
   376→   376→}

   396→   396→/**
   397→   397→ * Attempt generation with a single provider, with retries.
   398→   398→ * Returns the result or throws the last error.
   399→   399→ */
   400→   400→async function attemptWithRetries(
   401→   401→  options: AiCompletionOptions,
   402→   402→  config: AiProviderConfig,
   403→   403→  maxRetries: number = 2
   404→   404→): Promise<AiCompletionResult> {
   405→   405→  let lastError: Error | null = null;
   406→   406→  for (let attempt = 0; attempt <= maxRetries; attempt++) {
   407→   407→    try {
   408→   408→      return await tryProvider(options, config);
   409→   409→    } catch (error) {
   410→   410→      lastError = error instanceof Error ? error : new Error(String(error));
   411→   411→      const status = (lastError as Error & { status?: number }).status;
   412→   412→
   413→   413→      // Don't retry auth errors (invalid key)
   414→   414→      if (status && isAuthError(status)) throw lastError;
   415→   415→
   416→   416→      if (attempt < maxRetries) {
   417→   417→        const delay = Math.pow(2, attempt) * 1000;
   418→   418→        console.warn(
   419→   419→          `[AI] ${config.provider} attempt ${attempt + 1} failed, retrying in ${delay}ms...`,
   420→   420→          lastError.message
   421→   421→        );
   422→   422→        await new Promise((resolve) => setTimeout(resolve, delay));
   423→   423→      }
   424→   424→    }
   425→   425→  }
   426→   426→  throw lastError;
   427→   427→}

   430→   430→// Main entry point — auto-detects backend + failover
   431→   431→// ═══════════════════════════════════════════════════════════════
   432→   432→
   433→   433→/**
   434→   434→ * Generate a completion using the appropriate backend with failover.
   435→   435→ *
   436→   436→ * Priority:
   437→   437→ *  1. If providerConfig is passed (from client), use it
   438→   438→ *  2. If env vars define a provider, use them
   439→   439→ *  3. Default: try z.ai SDK
   440→   440→ *
   441→   441→ * Failover (inspired by freellmapi):
   442→   442→ *  - On retryable errors (429, 5xx), tries fallback providers in order
   443→   443→ *  - On auth errors (401, 403), immediately fails (no fallback)
   444→   444→ */
   445→   445→export async function generateCompletion(
   446→   446→  options: AiCompletionOptions
   447→   447→): Promise<AiCompletionResult> {
   448→   448→  const config = options.providerConfig || getDefaultConfig();
   449→   449→  const fallbackConfig = options.fallbackConfig;
   450→   450→
   451→   451→  // Inject hardcoded key if config has no apiKey
   452→   452→  function enrichWithHardcodedKey(c: AiProviderConfig): AiProviderConfig {
   453→   453→    if (!c.apiKey && c.provider !== "zai") {
   454→   454→      const hardcoded = getHardcodedKey(c.provider);
   455→   455→      if (hardcoded) {
   456→   456→        return { ...c, apiKey: hardcoded };
   457→   457→      }
   458→   458→    }
   459→   459→    return c;
   460→   460→  }
   461→   461→
   462→   462→  // Build the chain: primary + fallbacks
   463→   463→  const chain: AiProviderConfig[] = [enrichWithHardcodedKey(config)];
   464→   464→  if (fallbackConfig?.enabled && fallbackConfig.providers.length > 0) {
   465→   465→    for (const fb of fallbackConfig.providers) {
   466→   466→      chain.push(enrichWithHardcodedKey({
   467→   467→        provider: fb.provider,
   468→   468→        model: fb.model || config.model,
   469→   469→        baseUrl: fb.baseUrl,
   470→   470→      }));
   471→   471→    }
   472→   472→  }
   473→   473→
   474→   474→  let lastError: Error | null = null;
   475→   475→
   476→   476→  for (let i = 0; i < chain.length; i++) {
   477→   477→    const providerConfig = chain[i];
   478→   478→    const pid = providerConfig.provider;
   479→   479→
   480→   480→    // Circuit breaker check
   481→   481→    if (!canRequest(pid)) {
   482→   482→      console.warn(`[Circuit] Skipping ${pid} — circuit open (cooldown ${CIRCUIT_RESET_TIMEOUT_MS / 1000}s)`);
   483→   483→      if (i === chain.length - 1) {
   484→   484→        lastError = new Error(
   485→   485→          `Fournisseur IA "${pid}" temporairement indisponible (trop d'échecs consécutifs). ` +
   486→   486→          `Réessayez dans ${Math.ceil((CIRCUIT_RESET_TIMEOUT_MS - (Date.now() - getCircuit(pid).lastFailureTime)) / 1000)}s ou changez de fournisseur.`
   487→   487→        );
   488→   488→      }
   489→   489→      continue;
   490→   490→    }
   491→   491→
   492→   492→    try {
   493→   493→      const result = await attemptWithRetries(options, providerConfig, 2);
   494→   494→      recordSuccess(pid);
   495→   495→      if (i > 0) {
   496→   496→        console.log(`[AI] Failover: primary failed, served by ${pid}`);
   497→   497→      }
   498→   498→      return result;
   499→   499→    } catch (error) {
   500→   500→      lastError = error instanceof Error ? error : new Error(String(error));
   501→   501→      const status = (lastError as Error & { status?: number }).status;
   502→   502→
   503→   503→      recordFailure(pid);
   504→   504→
   505→   505→      // Auth errors are terminal — don't try fallbacks
   506→   506→      if (status && isAuthError(status)) throw lastError;
   507→   507→
   508→   508→      // Retryable or unknown error → try next in chain
   509→   509→      if (i < chain.length - 1) {
   510→   510→        console.warn(
   511→   511→          `[AI] Provider ${pid} failed (status=${status ?? "unknown"}), trying fallback ${i + 2}/${chain.length}...`
   512→   512→        );
   513→   513→      }
   514→   514→    }
   515→   515→  }
   516→   516→
   517→   517→  // All providers exhausted
   518→   518→  throw new Error(
   519→   519→    `Tous les fournisseurs IA ont échoué (chaîne de ${chain.length} fournisseurs). ` +
   520→   520→    `Dernière erreur: ${lastError?.message}`
   521→   521→  );
   522→   522→}

   557→   557→/**
   558→   558→ * Stream a completion using the appropriate backend with circuit breaker + failover.
   559→   559→ * Returns a Response with SSE content type.
   560→   560→ */
   561→   561→export async function generateCompletionStream(
   562→   562→  options: AiCompletionOptions
   563→   563→): Promise<Response> {
   564→   564→  const config = options.providerConfig || getDefaultConfig();
   565→   565→  const fallbackConfig = options.fallbackConfig;
   566→   566→
   567→   567→  function enrichWithHardcodedKey(c: AiProviderConfig): AiProviderConfig {
   568→   568→    if (!c.apiKey && c.provider !== "zai") {
   569→   569→      const hardcoded = getHardcodedKey(c.provider);
   570→   570→      if (hardcoded) return { ...c, apiKey: hardcoded };
   571→   571→    }
   572→   572→    return c;
   573→   573→  }
   574→   574→
   575→   575→  const chain: AiProviderConfig[] = [enrichWithHardcodedKey(config)];
   576→   576→  if (fallbackConfig?.enabled && fallbackConfig.providers.length > 0) {
   577→   577→    for (const fb of fallbackConfig.providers) {
   578→   578→      chain.push(enrichWithHardcodedKey({
   579→   579→        provider: fb.provider,
   580→   580→        model: fb.model || config.model,
   581→   581→        baseUrl: fb.baseUrl,
   582→   582→      }));
   583→   583→    }
   584→   584→  }
   585→   585→
   586→   586→  // Try providers in chain with circuit breaker
   587→   587→  for (let i = 0; i < chain.length; i++) {
   588→   588→    const providerConfig = chain[i];
   589→   589→    const pid = providerConfig.provider;
   590→   590→
   591→   591→    if (!canRequest(pid)) {
   592→   592→      console.warn(`[Circuit/Stream] Skipping ${pid} — circuit open`);
   593→   593→      if (i === chain.length - 1) {
   594→   594→        return new Response(
   595→   595→          formatSSE({ type: "error", error: `Fournisseur "${pid}" temporairement indisponible. Réessayez dans quelques secondes.` }),
   596→   596→          { headers: { "Content-Type": "text/event-stream", "Cache-Control": "no-cache" } }
   597→   597→        );
   598→   598→      }
   599→   599→      continue;
   600→   600→    }
   601→   601→
   602→   602→    try {
   603→   603→      const response = await streamWithProvider(options, providerConfig);
   604→   604→      recordSuccess(pid);
   605→   605→      if (i > 0) {
   606→   606→        console.log(`[AI/Stream] Failover: served by ${pid}`);
   607→   607→      }
   608→   608→      return response;
   609→   609→    } catch (error) {
   610→   610→      const err = error instanceof Error ? error : new Error(String(error));
   611→   611→      recordFailure(pid);
   612→   612→      const status = (err as Error & { status?: number }).status;
   613→   613→
   614→   614→      if (status && isAuthError(status)) {
   615→   615→        return new Response(
   616→   616→          formatSSE({ type: "error", error: err.message }),
   617→   617→          { headers: { "Content-Type": "text/event-stream", "Cache-Control": "no-cache" } }
   618→   618→        );
   619→   619→      }
   620→   620→
   621→   621→      if (i < chain.length - 1) {
   622→   622→        console.warn(`[AI/Stream] ${pid} failed, trying fallback ${i + 2}/${chain.length}...`);
   623→   623→      }
   624→   624→    }
   625→   625→  }
   626→   626→
   627→   627→  return new Response(
   628→   628→    formatSSE({ type: "error", error: "Tous les fournisseurs IA ont échoué." }),
   629→   629→    { headers: { "Content-Type": "text/event-stream", "Cache-Control": "no-cache" } }
   630→   630→  );
   631→   631→}

   633→   633→/**
   634→   634→ * Stream from a single provider (OpenAI SSE format)
   635→   635→ */
   636→   636→async function streamWithProvider(
   637→   637→  options: AiCompletionOptions,
   638→   638→  config: AiProviderConfig
   639→   639→): Promise<Response> {
   640→   640→  const backend = detectBackend(config.provider);
   641→   641→
   642→   642→  // zai SDK doesn't support streaming natively — fall back to non-streaming
   643→   643→  if (backend === "zai") {
   644→   644→    const result = await generateWithSDK(options);
   645→   645→    // Convert single response to a single-chunk stream
   646→   646→    return new Response(
   647→   647→      formatSSE({ type: "chunk", content: result.content, provider: result.provider, model: result.model }) +
   648→   648→      formatSSE({ type: "done", provider: result.provider, model: result.model }),
   649→   649→      { headers: { "Content-Type": "text/event-stream", "Cache-Control": "no-cache" } }
   650→   650→    );
   651→   651→  }
   652→   652→
   653→   653→  const baseUrl = getBaseUrl(config.provider, config.baseUrl);
   654→   654→  if (!baseUrl) throw new Error(`Aucune URL de base pour "${config.provider}".`);
   655→   655→
   656→   656→  const model = options.model || config.model || "gpt-4o-mini";
   657→   657→  const url = `${baseUrl}/chat/completions`;
   658→   658→  const headers: Record<string, string> = { "Content-Type": "application/json" };
   659→   659→
   660→   660→  if (!isKeylessProvider(config.provider) && config.apiKey) {
   661→   661→    if (isAnthropicFormat(config.provider)) {
   662→   662→      headers["x-api-key"] = config.apiKey;
   663→   663→      headers["anthropic-version"] = "2023-06-01";
   664→   664→      headers["anthropic-dangerous-direct-browser-access"] = "true";
   665→   665→    } else {
   666→   666→      headers["Authorization"] = `Bearer ${config.apiKey}`;
   667→   667→    }
   668→   668→  }
   669→   669→
   670→   670→  const extraHeaders = getProviderExtraHeaders(config.provider);
   671→   671→  Object.assign(headers, extraHeaders);
   672→   672→
   673→   673→  const systemMessage = options.messages.find((m) => m.role === "system");
   674→   674→  const nonSystemMessages = options.messages.filter((m) => m.role !== "system");
   675→   675→
   676→   676→  let body: Record<string, unknown>;
   677→   677→  if (isAnthropicFormat(config.provider) && systemMessage) {
   678→   678→    body = {
   679→   679→      model,
   680→   680→      max_tokens: options.maxTokens || 4096,
   681→   681→      temperature: options.temperature ?? 0.7,
   682→   682→      stream: true,
   683→   683→      system: systemMessage.content,
   684→   684→      messages: nonSystemMessages.map((m) => ({ role: m.role, content: m.content })),
   685→   685→    };
   686→   686→  } else {
   687→   687→    body = {
   688→   688→      model,
   689→   689→      temperature: options.temperature ?? 0.7,
   690→   690→      max_tokens: options.maxTokens || 4096,
   691→   691→      stream: true,
   692→   692→      messages: options.messages.map((m) => ({ role: m.role, content: m.content })),
   693→   693→    };
   694→   694→  }
   695→   695→
   696→   696→  // Create a TransformStream to convert SSE chunks into our format
   697→   697→  const { readable, writable } = new TransformStream<StreamChunk, string>({
   698→   698→    transform(chunk, controller) {
   699→   699→      controller.enqueue(formatSSE(chunk));
   700→   700→    },
   701→   701→  });
   702→   702→
   703→   703→  // Start fetching and parsing in the background
   704→   704→  (async () => {
   705→   705→    try {
   706→   706→      const fetchResponse = await fetch(url, {
   707→   707→        method: "POST",
   708→   708→        headers,
   709→   709→        body: JSON.stringify(body),
   710→   710→        signal: AbortSignal.timeout(REQUEST_TIMEOUT_MS),
   711→   711→      });
   712→   712→
   713→   713→      if (!fetchResponse.ok) {
   714→   714→        const apiErr = await buildApiError(fetchResponse, model);
   715→   715→        const writer = writable.getWriter();
   716→   716→        await writer.write({ type: "error", error: apiErr.message });
   717→   717→        await writer.close();
   718→   718→        return;
   719→   719→      }
   720→   720→
   721→   721→      const reader = fetchResponse.body?.getReader();
   722→   722→      if (!reader) {
   723→   723→        const writer = writable.getWriter();
   724→   724→        await writer.write({ type: "error", error: "Réponse vide du fournisseur." });
   725→   725→        await writer.close();
   726→   726→        return;
   727→   727→      }
   728→   728→
   729→   729→      const decoder = new TextDecoder();
   730→   730→      let buffer = "";
   731→   731→      const writer = writable.getWriter();
   732→   732→
   733→   733→      try {
   734→   734→        while (true) {
   735→   735→          const { done, value } = await reader.read();
   736→   736→          if (done) break;
   737→   737→
   738→   738→          buffer += decoder.decode(value, { stream: true });
   739→   739→          const lines = buffer.split("\n");
   740→   740→          buffer = lines.pop() || "";
   741→   741→
   742→   742→          for (const line of lines) {
   743→   743→            const trimmed = line.trim();
   744→   744→            if (!trimmed || !trimmed.startsWith("data: ")) continue;
   745→   745→            const data = trimmed.slice(6);
   746→   746→            if (data === "[DONE]") continue;
   747→   747→
   748→   748→            try {
   749→   749→              const parsed = JSON.parse(data) as Record<string, unknown>;
   750→   750→              const choices = parsed.choices as Array<Record<string, unknown>> | undefined;
   751→   751→              const delta = choices?.[0]?.delta as Record<string, unknown> | undefined;
   752→   752→              const content = String(delta?.content || "");
   753→   753→              if (content) {
   754→   754→                await writer.write({
   755→   755→                  type: "chunk",
   756→   756→                  content,
   757→   757→                  provider: config.provider,
   758→   758→                  model,
   759→   759→                });
   760→   760→              }
   761→   761→            } catch {
   762→   762→              // Skip malformed SSE data
   763→   763→            }
   764→   764→          }
   765→   765→        }
   766→   766→
   767→   767→        await writer.write({ type: "done", provider: config.provider, model });
   768→   768→      } finally {
   769→   769→        await writer.close();
   770→   770→      }
   771→   771→    } catch (error) {
   772→   772→      const writer = writable.getWriter();
   773→   773→      const msg = error instanceof Error ? error.message : "Erreur de streaming";
   774→   774→      await writer.write({ type: "error", error: msg });
   775→   775→      await writer.close();
   776→   776→    }
   777→   777→  })();
   778→   778→
   779→   779→  return new Response(readable, {
   780→   780→    headers: {
   781→   781→      "Content-Type": "text/event-stream",
   782→   782→      "Cache-Control": "no-cache",
   783→   783→      "Connection": "keep-alive",
   784→   784→    },
   785→   785→  });
   786→   786→}
