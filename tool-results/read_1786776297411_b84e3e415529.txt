    59→interface AgileStory {
    60→  id: string;
    61→  sprintId: string;
    62→  title: string;
    63→  description?: string | null;
    64→  status: string;
    65→  priority: string;
    66→  storyPoints?: number | null;
    67→  sortOrder: number;
    68→  createdAt: string;
    69→  updatedAt: string;
    70→}
    71→
    72→interface AgileSprint {
    73→  id: string;
    74→  phase: string;
    75→  title: string;
    76→  description?: string | null;
    77→  startDate?: string | null;
    78→  endDate?: string | null;
    79→  status: string;
    80→  sortOrder: number;
    81→  stories?: AgileStory[];
    82→  _count?: { stories: number };
    83→  createdAt: string;
    84→  updatedAt: string;
    85→}

    87→// ═══════════════════════════════════════
    88→// Constants
    89→// ═══════════════════════════════════════
    90→
    91→const COLUMNS = [
    92→  { id: "todo", label: "À faire", sublabel: "Backlog", icon: FolderKanban } as const,
    93→  { id: "in_progress", label: "En cours", sublabel: "En travaux", icon: Clock } as const,
    94→  { id: "in_review", label: "En revue", sublabel: "À vérifier", icon: Eye } as const,
    95→  { id: "done", label: "Terminé", sublabel: "Complété", icon: CheckCircle2 } as const,
    96→] as const;
    97→
    98→type ColumnId = (typeof COLUMNS)[number]["id"];
    99→
   100→const PHASE_LABELS: Record<string, string> = {
   101→  phase_0: "Cadrage",
   102→  phase_1: "Revue",
   103→  phase_2: "Rédaction",
   104→  phase_3: "Révision",
   105→  phase_4: "Finalisation",
   106→};
   107→
   108→const PRIORITY_LABELS: Record<string, { label: string; className: string }> = {
   109→  critical: { label: "Critique", className: "bg-red-500/15 text-red-700 dark:text-red-400 border-red-500/25" },
   110→  high: { label: "Haute", className: "bg-orange-500/15 text-orange-700 dark:text-orange-400 border-orange-500/25" },
   111→  medium: { label: "Moyenne", className: "bg-amber-500/15 text-amber-700 dark:text-amber-400 border-amber-500/25" },
   112→  low: { label: "Basse", className: "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/25" },
   113→};
   114→
   115→const PHASE_COLORS: Record<string, string> = {
   116→  phase_0: "bg-chart-4/15 text-chart-4 border-chart-4/25",
   117→  phase_1: "bg-chart-1/15 text-chart-1 border-chart-1/25",
   118→  phase_2: "bg-chart-2/15 text-chart-2 border-chart-2/25",
   119→  phase_3: "bg-chart-3/15 text-chart-3 border-chart-3/25",
   120→  phase_4: "bg-chart-5/15 text-chart-5 border-chart-5/25",
   121→};
   122→
   123→// Map 4-column UI status to 3-value DB status
   124→function toApiStatus(columnId: ColumnId): string {
   125→  if (columnId === "in_review") return "in_progress";
   126→  return columnId;
   127→}

   129→// ═══════════════════════════════════════
   130→// Component
   131→// ═══════════════════════════════════════
   132→
   133→export function FeuilleRouteAgilePage() {
   134→  const queryClient = useQueryClient();
   135→  const [selectedSprintId, setSelectedSprintId] = useState<string | null>(null);
   136→  const [createSprintOpen, setCreateSprintOpen] = useState(false);
   137→  const [createStoryOpen, setCreateStoryOpen] = useState(false);
   138→
   139→  // ─── Create Sprint form state ───
   140→  const [newSprintTitle, setNewSprintTitle] = useState("");
   141→  const [newSprintPhase, setNewSprintPhase] = useState("phase_2");
   142→  const [newSprintDesc, setNewSprintDesc] = useState("");
   143→
   144→  // ─── Create Story form state ───
   145→  const [newStoryTitle, setNewStoryTitle] = useState("");
   146→  const [newStoryDesc, setNewStoryDesc] = useState("");
   147→  const [newStoryPriority, setNewStoryPriority] = useState("medium");
   148→  const [newStoryPoints, setNewStoryPoints] = useState("");
   149→
   150→  // ─── Track which column each story is visually in (for "in_review" which has no DB status) ───
   151→  const [storyColumnMap, setStoryColumnMap] = useState<Record<string, ColumnId>>({});
   152→
   153→  // ─── Active tab ───
   154→  const [activeTab, setActiveTab] = useState("kanban");
   155→
   156→  // ═══════════════════════════════════════
   157→  // Queries
   158→  // ═══════════════════════════════════════
   159→
   160→  const { data: sprintsData, isLoading: sprintsLoading } = useQuery<{
   161→    data: AgileSprint[];
   162→  }>({ queryKey: ["sprints"], queryFn: () => fetch("/api/sprints").then((r) => r.json()) });
   163→
   164→  const sprints = sprintsData?.data ?? [];
   165→
   166→  // Auto-select first sprint if none selected
   167→  const activeSprintId = selectedSprintId ?? sprints[0]?.id ?? null;
   168→
   169→  const { data: activeSprintData, isLoading: sprintLoading } = useQuery<{
   170→    data: AgileSprint;
   171→  }>({
   172→    queryKey: ["sprint", activeSprintId],
   173→    queryFn: () => fetch(`/api/sprints/${activeSprintId}`).then((r) => r.json()),
   174→    enabled: !!activeSprintId,
   175→  });
   176→
   177→  const activeSprint = activeSprintData?.data ?? null;
   178→  const stories = activeSprint?.stories ?? [];
   179→
   180→  // ═══════════════════════════════════════
   181→  // Mutations
   182→  // ═══════════════════════════════════════
   183→
   184→  const createSprintMutation = useMutation({
   185→    mutationFn: async (body: { phase: string; title: string; description?: string }) => {
   186→      const res = await fetch("/api/sprints", {
   187→        method: "POST",
   188→        headers: { "Content-Type": "application/json" },
   189→        body: JSON.stringify(body),
   190→      });
   191→      if (!res.ok) throw new Error("Erreur lors de la création du sprint");
   192→      return res.json();
   193→    },
   194→    onSuccess: (data) => {
   195→      queryClient.invalidateQueries({ queryKey: ["sprints"] });
   196→      setSelectedSprintId(data.data.id);
   197→      setCreateSprintOpen(false);
   198→      setNewSprintTitle("");
   199→      setNewSprintDesc("");
   200→      toast.success("Sprint créé avec succès");
   201→    },
   202→    onError: () => toast.error("Impossible de créer le sprint"),
   203→  });
   204→
   205→  const createStoryMutation = useMutation({
   206→    mutationFn: async (body: {
   207→      title: string;
   208→      description?: string;
   209→      priority: string;
   210→      storyPoints?: number;
   211→    }) => {
   212→      if (!activeSprintId) throw new Error("Aucun sprint sélectionné");
   213→      const res = await fetch(`/api/sprints/${activeSprintId}/stories`, {
   214→        method: "POST",
   215→        headers: { "Content-Type": "application/json" },
   216→        body: JSON.stringify(body),
   217→      });
   218→      if (!res.ok) throw new Error("Erreur lors de la création de la story");
   219→      return res.json();
   220→    },
   221→    onSuccess: () => {
   222→      queryClient.invalidateQueries({ queryKey: ["sprint", activeSprintId] });
   223→      queryClient.invalidateQueries({ queryKey: ["sprints"] });
   224→      setCreateStoryOpen(false);
   225→      setNewStoryTitle("");
   226→      setNewStoryDesc("");
   227→      setNewStoryPriority("medium");
   228→      setNewStoryPoints("");
   229→      toast.success("Story ajoutée au sprint");
   230→    },
   231→    onError: () => toast.error("Impossible de créer la story"),
   232→  });
   233→
   234→  const updateStoryMutation = useMutation({
   235→    mutationFn: async ({ id, status }: { id: string; status: string }) => {
   236→      const res = await fetch(`/api/stories/${id}`, {
   237→        method: "PUT",
   238→        headers: { "Content-Type": "application/json" },
   239→        body: JSON.stringify({ status }),
   240→      });
   241→      if (!res.ok) throw new Error("Erreur lors de la mise à jour");
   242→      return res.json();
   243→    },
   244→    onSuccess: () => {
   245→      queryClient.invalidateQueries({ queryKey: ["sprint", activeSprintId] });
   246→      queryClient.invalidateQueries({ queryKey: ["sprints"] });
   247→    },
   248→    onError: () => toast.error("Impossible de déplacer la story"),
   249→  });
   250→
   251→  const deleteStoryMutation = useMutation({
   252→    mutationFn: async (id: string) => {
   253→      const res = await fetch(`/api/stories/${id}`, { method: "DELETE" });
   254→      if (!res.ok) throw new Error("Erreur lors de la suppression");
   255→      return res.json();
   256→    },
   257→    onSuccess: () => {
   258→      queryClient.invalidateQueries({ queryKey: ["sprint", activeSprintId] });
   259→      queryClient.invalidateQueries({ queryKey: ["sprints"] });
   260→      toast.success("Story supprimée");
   261→    },
   262→    onError: () => toast.error("Impossible de supprimer la story"),
   263→  });
   264→
   265→  const deleteSprintMutation = useMutation({
   266→    mutationFn: async (id: string) => {
   267→      const res = await fetch(`/api/sprints/${id}`, { method: "DELETE" });
   268→      if (!res.ok) throw new Error("Erreur lors de la suppression");
   269→      return res.json();
   270→    },
   271→    onSuccess: (_, deletedId) => {
   272→      queryClient.invalidateQueries({ queryKey: ["sprints"] });
   273→      if (selectedSprintId === deletedId) setSelectedSprintId(null);
   274→      toast.success("Sprint supprimé");
   275→    },
   276→    onError: () => toast.error("Impossible de supprimer le sprint"),
   277→  });
   278→
   279→  const generateSprintMutation = useMutation({
   280→    mutationFn: async () => {
   281→      const res = await fetch("/api/ai-writing", {
   282→        method: "POST",
   283→        headers: { "Content-Type": "application/json" },
   284→        body: JSON.stringify({
   285→          mode: "sprint_planning",
   286→          input: `Phase de thèse: ${PHASE_LABELS[activeSprint?.phase ?? "phase_2"] ?? "Rédaction"}. Sprint actuel: ${activeSprint?.title ?? ""}. Stories existantes: ${stories.map((s) => s.title).join(", ") || "aucune"}. Génère un plan de sprint avec 4-6 stories utilisateur pour la thèse.`,
   287→        }),
   288→      });
   289→      if (!res.ok) throw new Error("Erreur IA");
   290→      return res.json();
   291→    },
   292→    onSuccess: (data) => {
   293→      toast.success("Suggestion IA générée", {
   294→        description: "Consultez les suggestions dans la zone de texte.",
   295→      });
   296→      setAiSuggestion(data.data?.result ?? data.result ?? "Aucune suggestion générée.");
   297→    },
   298→    onError: () => toast.error("Impossible de générer des suggestions IA"),
   299→  });
   300→
   301→  const [aiSuggestion, setAiSuggestion] = useState("");
   302→
   303→  // ═══════════════════════════════════════
   304→  // Column assignment logic
   305→  // ═══════════════════════════════════════
   306→
   307→  const getStoryColumn = useCallback(
   308→    (story: AgileStory): ColumnId => {
   309→      // Check local override first
   310→      if (storyColumnMap[story.id]) return storyColumnMap[story.id];
   311→      // Default: map DB status to column
   312→      return story.status as ColumnId;
   313→    },
   314→    [storyColumnMap]
   315→  );
   316→
   317→  const columnStories = useMemo(() => {
   318→    const map: Record<ColumnId, AgileStory[]> = {
   319→      todo: [],
   320→      in_progress: [],
   321→      in_review: [],
   322→      done: [],
   323→    };
   324→    for (const story of stories) {
   325→      const col = getStoryColumn(story);
   326→      map[col].push(story);
   327→    }
   328→    return map;
   329→  }, [stories, getStoryColumn]);
   330→
   331→  // ═══════════════════════════════════════
   332→  // Move story between columns
   333→  // ═══════════════════════════════════════
   334→
   335→  const moveStory = useCallback(
   336→    (storyId: string, direction: "left" | "right") => {
   337→      const story = stories.find((s) => s.id === storyId);
   338→      if (!story) return;
   339→
   340→      const currentCol = getStoryColumn(story);
   341→      const currentIndex = COLUMNS.findIndex((c) => c.id === currentCol);
   342→      const newIndex = direction === "right" ? currentIndex + 1 : currentIndex - 1;
   343→
   344→      if (newIndex < 0 || newIndex >= COLUMNS.length) return;
   345→
   346→      const targetColumn = COLUMNS[newIndex].id;
   347→      const apiStatus = toApiStatus(targetColumn);
   348→
   349→      // Update local column map
   350→      setStoryColumnMap((prev) => ({ ...prev, [storyId]: targetColumn }));
   351→
   352→      // Persist to DB
   353→      updateStoryMutation.mutate({ id: storyId, status: apiStatus });
   354→    },
   355→    [stories, getStoryColumn, updateStoryMutation]
   356→  );
   357→
   358→  // ═══════════════════════════════════════
   359→  // Statistics
   360→  // ═══════════════════════════════════════
   361→
   362→  const stats = useMemo(() => {
   363→    const total = stories.length;
   364→    const done = stories.filter((s) => s.status === "done").length;
   365→    const inProgress = stories.filter((s) => s.status === "in_progress").length;
   366→    const totalPoints = stories.reduce((sum, s) => sum + (s.storyPoints ?? 0), 0);
   367→    const avgPoints = total > 0 ? Math.round((totalPoints / total) * 10) / 10 : 0;
   368→    const progressPercent = total > 0 ? Math.round((done / total) * 100) : 0;
   369→    return { total, done, inProgress, avgPoints, totalPoints, progressPercent };
   370→  }, [stories]);
   371→
   372→  // ═══════════════════════════════════════
   373→  // Handlers
   374→  // ═══════════════════════════════════════
   375→
   376→  const handleCreateSprint = () => {
   377→    if (!newSprintTitle.trim()) {
   378→      toast.error("Le titre du sprint est requis");
   379→      return;
   380→    }
   381→    createSprintMutation.mutate({
   382→      phase: newSprintPhase,
   383→      title: newSprintTitle.trim(),
   384→      description: newSprintDesc.trim() || undefined,
   385→    });
   386→  };
   387→
   388→  const handleCreateStory = () => {
   389→    if (!newStoryTitle.trim()) {
   390→      toast.error("Le titre de la story est requis");
   391→      return;
   392→    }
   393→    createStoryMutation.mutate({
   394→      title: newStoryTitle.trim(),
   395→      description: newStoryDesc.trim() || undefined,
   396→      priority: newStoryPriority,
   397→      storyPoints: newStoryPoints ? parseInt(newStoryPoints, 10) : undefined,
   398→    });
   399→  };
   400→
   401→  // ═══════════════════════════════════════
   402→  // Render helpers
   403→  // ═══════════════════════════════════════
   404→
   405→  const renderStoryCard = (story: AgileStory, colIndex: number) => {
   406→    const priority = PRIORITY_LABELS[story.priority] ?? PRIORITY_LABELS.medium;
   407→    const phase = activeSprint?.phase ?? "phase_2";
   408→    const phaseColor = PHASE_COLORS[phase] ?? "";
   409→    const phaseLabel = PHASE_LABELS[phase] ?? "Rédaction";
   410→
   411→    return (
   412→      <Card key={story.id} className="p-3 flex flex-col gap-2 group hover:shadow-md transition-shadow">
   413→        <div className="flex items-start justify-between gap-2">
   414→          <p className="text-sm font-medium leading-snug flex-1">{story.title}</p>
   415→          <Button
   416→            variant="ghost"
   417→            size="icon"
   418→            className="h-6 w-6 opacity-0 group-hover:opacity-100 transition-opacity shrink-0"
   419→            onClick={() => deleteStoryMutation.mutate(story.id)}
   420→          >
   421→            <Trash2 className="h-3 w-3 text-muted-foreground" />
   422→          </Button>
   423→        </div>
   424→        {story.description && (
   425→          <p className="text-xs text-muted-foreground line-clamp-2">{story.description}</p>
   426→        )}
   427→        <div className="flex items-center gap-1.5 flex-wrap">
   428→          <Badge variant="outline" className={`text-[10px] px-1.5 py-0 ${priority.className}`}>
   429→            {priority.label}
   430→          </Badge>
   431→          <Badge variant="outline" className={`text-[10px] px-1.5 py-0 ${phaseColor}`}>
   432→            {phaseLabel}
   433→          </Badge>
   434→          {story.storyPoints != null && story.storyPoints > 0 && (
   435→            <Badge variant="secondary" className="text-[10px] px-1.5 py-0">
   436→              {story.storyPoints} pt{story.storyPoints > 1 ? "s" : ""}
   437→            </Badge>
   438→          )}
   439→        </div>
   440→        <div className="flex items-center gap-1 justify-end pt-1">
   441→          {colIndex > 0 && (
   442→            <Button
   443→              variant="ghost"
   444→              size="icon"
   445→              className="h-6 w-6"
   446→              onClick={() => moveStory(story.id, "left")}
   447→              title={"Déplacer vers " + COLUMNS[colIndex - 1].label}
   448→            >
   449→              <ChevronLeft className="h-3 w-3" />
   450→            </Button>
   451→          )}
   452→          {colIndex < COLUMNS.length - 1 && (
   453→            <Button
   454→              variant="ghost"
   455→              size="icon"
   456→              className="h-6 w-6"
   457→              onClick={() => moveStory(story.id, "right")}
   458→              title={"Déplacer vers " + COLUMNS[colIndex + 1].label}
   459→            >
   460→              <ChevronRight className="h-3 w-3" />
   461→            </Button>
   462→          )}
   463→        </div>
   464→      </Card>
   465→    );
   466→  };
   467→
   468→  // ═══════════════════════════════════════
   469→  // Loading state
   470→  // ═══════════════════════════════════════
   471→
   472→  if (sprintsLoading) {
   473→    return (
   474→      <div className="max-w-6xl mx-auto flex flex-col gap-6 p-6">
   475→        <div className="flex items-center gap-3">
   476→          <Skeleton className="h-8 w-8 rounded-lg" />
   477→          <div className="flex flex-col gap-1">
   478→            <Skeleton className="h-6 w-48" />
   479→            <Skeleton className="h-4 w-72" />
   480→          </div>
   481→        </div>
   482→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   483→          {Array.from({ length: 4 }).map((_, i) => (
   484→            <Skeleton key={i} className="h-24 rounded-lg" />
   485→          ))}
   486→        </div>
   487→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   488→          {Array.from({ length: 4 }).map((_, i) => (
   489→            <Skeleton key={i} className="h-64 rounded-lg" />
   490→          ))}
   491→        </div>
   492→      </div>
   493→    );
   494→  }
   495→
   496→  // ═══════════════════════════════════════
   497→  // Empty state — no sprints
   498→  // ═══════════════════════════════════════
   499→
   500→  if (sprints.length === 0) {
   501→    return (
   502→      <div className="max-w-6xl mx-auto flex flex-col gap-6 p-6">
   503→        <div className="flex items-center gap-3">
   504→          <div className="flex items-center justify-center h-8 w-8 rounded-lg bg-chart-1/15">
   505→            <Route className="h-4 w-4 text-chart-1" />
   506→          </div>
   507→          <div>
   508→            <h1 className="text-xl font-semibold">Feuille de Route Agile</h1>
   509→            <p className="text-sm text-muted-foreground">
   510→              Planifiez et suivez la progression de votre thèse avec la méthode agile
   511→            </p>
   512→          </div>
   513→        </div>
   514→        <Separator />
   515→        <Card className="p-8 flex flex-col items-center justify-center gap-4 text-center">
   516→          <div className="flex items-center justify-center h-12 w-12 rounded-full bg-muted">
   517→            <KanbanSquare className="h-6 w-6 text-muted-foreground" />
   518→          </div>
   519→          <div>
   520→            <h3 className="font-medium">Aucun sprint créé</h3>
   521→            <p className="text-sm text-muted-foreground mt-1">
   522→              Commencez par créer votre premier sprint pour organiser le travail de votre thèse.
   523→            </p>
   524→          </div>
   525→          <Dialog open={createSprintOpen} onOpenChange={setCreateSprintOpen}>
   526→            <DialogTrigger asChild>
   527→              <Button>
   528→                <Plus className="h-4 w-4 mr-2" />
   529→                Créer un sprint
   530→              </Button>
   531→            </DialogTrigger>
   532→            {renderCreateSprintDialog()}
   533→          </Dialog>
   534→        </Card>
   535→      </div>
