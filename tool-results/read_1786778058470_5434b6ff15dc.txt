    38→/* ═══════════════════════════════════════════════════════════════════════
    39→   TYPES

    41→
    42→interface FormData {
    43→  topic: string;
    44→  discipline: string;
    45→  keywords: string;
    46→  questions: string;
    47→}
    48→
    49→interface AnalysisResult {
    50→  overview: string;
    51→  authors: string;
    52→  gaps: string;
    53→  directions: string;
    54→}
    55→
    56→interface KnowledgeNode {
    57→  id: string;
    58→  label: string;
    59→  children?: KnowledgeNode[];
    60→}
    61→
    62→interface PositioningData {
    63→  originality: string;
    64→  theoreticalAnchoring: string;
    65→  methodologicalApproach: string;
    66→  expectedContribution: string;
    67→}
    68→
    69→/* ═══════════════════════════════════════════════════════════════════════
    70→   MAIN COMPONENT
    71→   ═══════════════════════════════════════════════════════════════════════ */
    72→
    73→export function AnalyseChampRecherchePage() {
    74→  // Form state
    75→  const [form, setForm] = useState<FormData>({
    76→    topic: "",
    77→    discipline: "",
    78→    keywords: "",
    79→    questions: "",
    80→  });
    81→
    82→  // Analysis state
    83→  const [isLoading, setIsLoading] = useState(false);
    84→  const [error, setError] = useState<string | null>(null);
    85→  const [analysis, setAnalysis] = useState<AnalysisResult | null>(null);
    86→  const [copied, setCopied] = useState(false);
    87→
    88→  // Knowledge map state
    89→  const [knowledgeMap, setKnowledgeMap] = useState<KnowledgeNode[]>([]);
    90→  const [newNodeLabel, setNewNodeLabel] = useState("");
    91→  const [newChildLabel, setNewChildLabel] = useState("");
    92→
    93→  // Positioning state
    94→  const [positioning, setPositioning] = useState<PositioningData>({
    95→    originality: "",
    96→    theoreticalAnchoring: "",
    97→    methodologicalApproach: "",
    98→    expectedContribution: "",
    99→  });
   100→  const [isPositioningLoading, setIsPositioningLoading] = useState(false);
   101→
   102→  // Literature landscape (derived from analysis)
   103→  const [landscapeItems, setLandscapeItems] = useState<
   104→    { title: string; description: string; type: string }[]
   105→  >([]);
   106→
   107→  const updateForm = (field: keyof FormData, value: string) => {
   108→    setForm((prev) => ({ ...prev, [field]: value }));
   109→  };
   110→
   111→  /* ── AI Analysis ──────────────────────────────────────────── */
   112→
   113→  const runAnalysis = useCallback(async () => {
   114→    if (!form.topic.trim()) {
   115→      toast({
   116→        title: "Champ requis",
   117→        description: "Veuillez saisir un sujet de recherche.",
   118→      });
   119→      return;
   120→    }
   121→
   122→    setIsLoading(true);
   123→    setError(null);
   124→
   125→    try {
   126→      const contextParts = [];
   127→      if (form.discipline) contextParts.push(`Discipline : ${form.discipline}`);
   128→      if (form.keywords) contextParts.push(`Mots-clés : ${form.keywords}`);
   129→      if (form.questions) contextParts.push(`Questions de recherche : ${form.questions}`);
   130→
   131→      const prompt = `Analyse en profondeur le champ de recherche suivant :
   132→
   133→Sujet : ${form.topic}
   134→${contextParts.length > 0 ? `\nContexte additionnel :\n${contextParts.join("\n")}` : ""}
   135→
   136→Fournis une analyse structurée avec les 4 sections suivantes, chacune clairement séparée par le marqueur ##SECTION## :
   137→
   138→##SECTION## ÉTAT DE L'ART
   139→Fais une synthèse de l'état de l'art sur ce sujet. Présente les grands courants, les débats majeurs et l'évolution chronologique du champ. 3-4 paragraphes.
   140→
   141→##SECTION## AUTEURS ET THÉORIES
   142→Identifie les auteurs fondateurs et contemporains les plus importants. Présente les théories clés du champ et leurs relations. 3-4 paragraphes.
   143→
   144→##SECTION## LACUNES IDENTIFIÉES
   145→Identifie les lacunes, contradictions et zones d'ombre dans la littérature existante. Quels aspects restent peu explorés ? 2-3 paragraphes.
   146→
   147→##SECTION## PISTES DE RECHERCHE
   148→Suggère des directions de recherche prometteuses, en lien avec les lacunes identifiées. Formule des questions de recherche potentielles. 2-3 paragraphes.`;
   149→
   150→      const res = await fetch("/api/ai-writing", {
   151→        method: "POST",
   152→        headers: { "Content-Type": "application/json" },
   153→        body: JSON.stringify({
   154→          mode: "literature-review",
   155→          prompt,
   156→          context: `Analyse du champ de recherche pour une thèse. Sujet : ${form.topic}. Discipline : ${form.discipline || "non précisée"}.`,
   157→        }),
   158→      });
   159→
   160→      if (!res.ok) throw new Error("Erreur lors de la génération");
   161→      const json = await res.json();
   162→      const content = json.data.content as string;
   163→
   164→      // Parse the 4 sections from the response
   165→      const sections = content.split("##SECTION##");
   166→      const parsed: AnalysisResult = {
   167→        overview: "",
   168→        authors: "",
   169→        gaps: "",
   170→        directions: "",
   171→      };
   172→
   173→      const sectionTitles = [
   174→        "ÉTAT DE L'ART",
   175→        "AUTEURS ET THÉORIES",
   176→        "LACUNES IDENTIFIÉES",
   177→        "PISTES DE RECHERCHE",
   178→      ];
   179→
   180→      for (let i = 1; i < sections.length && i <= 4; i++) {
   181→        let text = sections[i].trim();
   182→        // Remove the section title from the beginning
   183→        for (const title of sectionTitles) {
   184→          if (text.startsWith(title)) {
   185→            text = text.slice(title.length).trim();
   186→            break;
   187→          }
   188→        }
   189→        if (i === 1) parsed.overview = text;
   190→        else if (i === 2) parsed.authors = text;
   191→        else if (i === 3) parsed.gaps = text;
   192→        else if (i === 4) parsed.directions = text;
   193→      }
   194→
   195→      setAnalysis(parsed);
   196→
   197→      // Auto-generate knowledge map from authors section
   198→      generateKnowledgeMap(parsed.authors);
   199→
   200→      // Auto-generate landscape items from overview
   201→      generateLandscape(parsed.overview);
   202→    } catch (err) {
   203→      setError(
   204→        err instanceof Error ? err.message : "Erreur lors de l'analyse"
   205→      );
   206→    } finally {
   207→      setIsLoading(false);
   208→    }
   209→  }, [form]);
   210→
   211→  /* ── Knowledge Map ──────────────────────────────────────────── */
   212→
   213→  const generateKnowledgeMap = (authorsText: string) => {
   214→    // Extract key concepts from the AI text
   215→    const lines = authorsText.split("\n").filter((l) => l.trim().length > 0);
   216→    const concepts = lines
   217→      .filter((l) => l.match(/^[•\-–—●]/) || l.match(/^\d+\./))
   218→      .map((l) => l.replace(/^[•\-–—●\d.)\s]+/, "").trim())
   219→      .filter((l) => l.length > 5 && l.length < 100);
   220→
   221→    if (concepts.length === 0) {
   222→      // Fallback: create nodes from paragraphs
   223→      const paragraphs = authorsText
   224→        .split(/\n\n/)
   225→        .map((p) => p.trim())
   226→        .filter((p) => p.length > 20);
   227→      setKnowledgeMap(
   228→        paragraphs.slice(0, 6).map((p, i) => ({
   229→          id: `node-${i}`,
   230→          label: p.length > 60 ? p.substring(0, 57) + "..." : p,
   231→        }))
   232→      );
   233→      return;
   234→    }
   235→
   236→    // Group concepts into a tree structure
   237→    const rootNodes: KnowledgeNode[] = [];
   238→    const groupSize = Math.max(2, Math.ceil(concepts.length / 4));
   239→
   240→    for (let i = 0; i < concepts.length; i += groupSize) {
   241→      const group = concepts.slice(i, i + groupSize);
   242→      rootNodes.push({
   243→        id: `node-${i}`,
   244→        label:
   245→          group[0].length > 60
   246→            ? group[0].substring(0, 57) + "..."
   247→            : group[0],
   248→        children: group.slice(1).map((c, j) => ({
   249→          id: `node-${i}-${j}`,
   250→          label: c.length > 50 ? c.substring(0, 47) + "..." : c,
   251→        })),
   252→      });
   253→    }
   254→
   255→    setKnowledgeMap(rootNodes);
   256→  };
   257→
   258→  const addRootNode = () => {
   259→    if (!newNodeLabel.trim()) return;
   260→    setKnowledgeMap((prev) => [
   261→      ...prev,
   262→      {
   263→        id: `custom-${Date.now()}`,
   264→        label: newNodeLabel.trim(),
   265→        children: [],
   266→      },
   267→    ]);
   268→    setNewNodeLabel("");
   269→  };
   270→
   271→  const addChildNode = (parentId: string) => {
   272→    if (!newChildLabel.trim()) return;
   273→    setKnowledgeMap((prev) =>
   274→      prev.map((node) =>
   275→        node.id === parentId
   276→          ? {
   277→              ...node,
   278→              children: [
   279→                ...(node.children || []),
   280→                {
   281→                  id: `child-${Date.now()}`,
   282→                  label: newChildLabel.trim(),
   283→                },
   284→              ],
   285→            }
   286→          : node
   287→      )
   288→    );
   289→    setNewChildLabel("");
   290→  };
   291→
   292→  const removeNode = (nodeId: string) => {
   293→    setKnowledgeMap((prev) => prev.filter((n) => n.id !== nodeId));
   294→  };
   295→
   296→  const removeChildNode = (parentId: string, childId: string) => {
   297→    setKnowledgeMap((prev) =>
   298→      prev.map((node) =>
   299→        node.id === parentId
   300→          ? {
   301→              ...node,
   302→              children: (node.children || []).filter((c) => c.id !== childId),
   303→            }
   304→          : node
   305→      )
   306→    );
   307→  };
   308→
   309→  /* ── Literature Landscape ──────────────────────────────────────── */
   310→
   311→  const generateLandscape = (overviewText: string) => {
   312→    const paragraphs = overviewText
   313→      .split(/\n\n/)
   314→      .map((p) => p.trim())
   315→      .filter((p) => p.length > 30);
   316→
   317→    const types = ["Courant", "Thématique", "Débat", "Perspective", "Approche"];
   318→
   319→    const items = paragraphs.slice(0, 6).map((p, i) => {
   320→      const firstSentence = p.split(".")[0] + ".";
   321→      return {
   322→        title:
   323→          firstSentence.length > 60
   324→            ? firstSentence.substring(0, 57) + "..."
   325→            : firstSentence,
   326→        description:
   327→          p.length > 200 ? p.substring(0, 197) + "..." : p,
   328→        type: types[i % types.length],
   329→      };
   330→    });
   331→
   332→    setLandscapeItems(items);
   333→  }, []);
   334→
   335→  /* ── Research Positioning AI ────────────────────────────────────── */
   336→
   337→  const generatePositioning = useCallback(async () => {
   338→    if (!form.topic.trim()) {
   339→      toast({
   340→        title: "Analyse requise",
   341→        description: "Lancez d'abord l'analyse du champ de recherche.",
   342→      });
   343→      return;
   344→    }
   345→
   346→    setIsPositioningLoading(true);
   347→    try {
   348→      const res = await fetch("/api/ai-writing", {
   349→        method: "POST",
   350→        headers: { "Content-Type": "application/json" },
   351→        body: JSON.stringify({
   352→          mode: "theory",
   353→          prompt: `Aide un doctorant à articuler son positionnement de recherche dans le champ suivant :
   354→
   355→Sujet : ${form.topic}
   356→Discipline : ${form.discipline || "non précisée"}
   357→Mots-clés : ${form.keywords || "non précisés"}
   358→Questions de recherche : ${form.questions || "non précisées"}
   359→
   360→${analysis ? `Analyse préalable du champ :\n- Lacunes : ${analysis.gaps}\n- Pistes : ${analysis.directions}` : ""}
   361→
   362→Génère 4 paragraphes structurés, chacun séparé par ##SECTION## :
   363→
   364→##SECTION## ORIGINALITÉ
   365→Aide à formuler ce qui rend ce travail original par rapport à l'existant. 1 paragraphe.
   366→
   367→##SECTION## ANCRAGE THÉORIQUE
   368→Suggère un cadre théorique pertinent pour ancrer cette recherche. 1 paragraphe.
   369→
   370→##SECTION## APPROCHE MÉTHODOLOGIQUE
   371→Propose une approche méthodologique adaptée au sujet et aux lacunes identifiées. 1 paragraphe.
   372→
   373→##SECTION## CONTRIBUTION ATTENDUE
   374→Aide à formuler la contribution scientifique attendue de cette thèse. 1 paragraphe.`,
   375→          context: `Positionnement de recherche doctoral. Sujet : ${form.topic}`,
   376→        }),
   377→      });
   378→
   379→      if (!res.ok) throw new Error("Erreur lors de la génération");
   380→      const json = await res.json();
   381→      const content = json.data.content as string;
   382→
   383→      const sections = content.split("##SECTION##");
   384→      const parsed: PositioningData = {
   385→        originality: "",
   386→        theoreticalAnchoring: "",
   387→        methodologicalApproach: "",
   388→        expectedContribution: "",
   389→      };
   390→
   391→      const titles = [
   392→        "ORIGINALITÉ",
   393→        "ANCRAGE THÉORIQUE",
   394→        "APPROCHE MÉTHODOLOGIQUE",
   395→        "CONTRIBUTION ATTENDUE",
   396→      ];
   397→
   398→      for (let i = 1; i < sections.length && i <= 4; i++) {
   399→        let text = sections[i].trim();
   400→        for (const title of titles) {
   401→          if (text.startsWith(title)) {
   402→            text = text.slice(title.length).trim();
   403→            break;
   404→          }
   405→        }
   406→        if (i === 1) parsed.originality = text;
   407→        else if (i === 2) parsed.theoreticalAnchoring = text;
   408→        else if (i === 3) parsed.methodologicalApproach = text;
   409→        else if (i === 4) parsed.expectedContribution = text;
   410→      }
   411→
   412→      setPositioning(parsed);
   413→      toast({
   414→        title: "Positionnement généré",
   415→        description: "Le positionnement de recherche a été généré avec succès.",
   416→      });
   417→    } catch {
   418→      toast({
   419→        title: "Erreur",
   420→        description: "Impossible de générer le positionnement. Réessayez.",
   421→      });
   422→    } finally {
   423→      setIsPositioningLoading(false);
   424→    }
   425→  }, [form, analysis]);
   426→
   427→  /* ── Export ──────────────────────────────────────────────── */
   428→
   429→  const exportReport = useCallback(async () => {
   430→    if (!analysis) {
   431→      toast({
   432→        title: "Rien à exporter",
   433→        description: "Lancez d'abord une analyse du champ de recherche.",
   434→      });
   435→      return;
   436→    }
   437→
   438→    const mapText = knowledgeMap
   439→      .map((node) => {
   440→        const children = (node.children || [])
   441→          .map((c) => `    └── ${c.label}`)
   442→          .join("\n");
   443→        return `├── ${node.label}${children ? `\n${children}` : ""}`;
   444→      })
   445→      .join("\n");
   446→
   447→    const landscapeText = landscapeItems
   448→      .map((item) => `- [${item.type}] ${item.title}\n  ${item.description}`)
   449→      .join("\n\n");
   450→
   451→    const positioningText = positioning.originality
   452→      ? `### Positionnement de recherche
   453→
   454→**Originalité :**
   455→${positioning.originality}
   456→
   457→**Ancrage théorique :**
   458→${positioning.theoreticalAnchoring}
   459→
   460→**Approche méthodologique :**
   461→${positioning.methodologicalApproach}
   462→
   463→**Contribution attendue :**
   464→${positioning.expectedContribution}`
   465→      : "";
   466→
   467→    const report = `# Analyse du champ de recherche
   468→
   469→**Sujet :** ${form.topic}
   470→**Discipline :** ${form.discipline || "Non précisée"}
   471→**Mots-clés :** ${form.keywords || "Non précisés"}
   472→**Questions de recherche :** ${form.questions || "Non précisées"}
   473→
   474→---
   475→
   476→## État de l'art
   477→
   478→${analysis.overview}
   479→
   480→## Auteurs et théories clés
   481→
   482→${analysis.authors}
   483→
   484→## Lacunes identifiées
   485→
   486→${analysis.gaps}
   487→
   488→## Pistes de recherche
   489→
   490→${analysis.directions}
   491→
   492→---
   493→
   494→## Carte des connaissances
   495→
   496→${form.topic}
   497→${mapText}
   498→
   499→---
   500→
   501→## Paysage littéraire
   502→
   503→${landscapeText}
   504→
   505→${positioningText ? `\n---\n\n${positioningText}` : ""}
   506→
   507→---
   508→*Généré par ThesisFrame — Analyse du champ de recherche*`;
   509→
   510→    try {
   511→      await navigator.clipboard.writeText(report);
   512→      setCopied(true);
   513→      toast({
   514→        title: "Rapport copié",
   515→        description: "Le rapport d'analyse a été copié dans le presse-papiers.",
   516→      });
   517→      setTimeout(() => setCopied(false), 2000);
   518→    } catch {
   519→      toast({
   520→        title: "Erreur",
   521→        description: "Impossible de copier dans le presse-papiers.",
   522→      });
   523→    }
   524→  }, [analysis, form, knowledgeMap, landscapeItems, positioning]);
   525→
   526→  const resetForm = () => {
   527→    setForm({ topic: "", discipline: "", keywords: "", questions: "" });
   528→    setAnalysis(null);
   529→    setKnowledgeMap([]);
   530→    setLandscapeItems([]);
   531→    setPositioning({
   532→      originality: "",
   533→      theoreticalAnchoring: "",
   534→      methodologicalApproach: "",
   535→      expectedContribution: "",
   536→    });
   537→    setError(null);
   538→  };
   539→
   540→  /* ═══════════════════════════════════════════════════════════════════════
   541→     RENDER
   542→     ═══════════════════════════════════════════════════════════════════════ */
   543→
   544→  return (
   545→    <div className="max-w-6xl mx-auto flex flex-col gap-6 p-6">
   546→      {/* ── Header ──────────────────────────────────────────── */}
   547→      <div className="flex items-start gap-4">
   548→        <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-chart-3/10">
   549→          <Compass className="h-5 w-5 text-chart-3" />
   550→        </div>
   551→        <div>
   552→          <h1 className="text-2xl font-bold tracking-tight">
   553→            Analyse du champ de recherche
   554→          </h1>
   555→          <p className="text-sm text-muted-foreground">
   556→            Cartographiez votre champ, identifiez les lacunes et positionnez
   557→            votre recherche grâce à l&apos;IA
   558→          </p>
   559→        </div>
   560→      </div>
   561→
   562→      {/* ── Form Card ────────────────────────────────────────── */}
   563→      <Card>
   564→        <CardHeader>
   565→          <CardTitle className="text-base flex items-center gap-2">
   566→            <Search className="h-4 w-4 text-chart-3" />
   567→            Définition du champ de recherche
   568→          </CardTitle>
   569→          <CardDescription>
   570→            Décrivez votre sujet pour lancer l&apos;analyse approfondie du champ
   571→          </CardDescription>
   572→        </CardHeader>
   573→        <CardContent className="flex flex-col gap-4">
   574→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   575→            <div className="flex flex-col gap-2">
   576→              <Label htmlFor="topic">Sujet de recherche *</Label>
   577→              <Input
   578→                id="topic"
   579→                placeholder="Ex : L'impact de l'IA générative sur l'enseignement supérieur"
   580→                value={form.topic}
   581→                onChange={(e) => updateForm("topic", e.target.value)}
   582→              />
   583→            </div>
   584→            <div className="flex flex-col gap-2">
   585→              <Label htmlFor="discipline">Discipline</Label>
   586→              <Input
   587→                id="discipline"
   588→                placeholder="Ex : Sciences de l'éducation, Sociologie..."
   589→                value={form.discipline}
   590→                onChange={(e) => updateForm("discipline", e.target.value)}
   591→              />
   592→            </div>
   593→          </div>
   594→          <div className="flex flex-col gap-2">
   595→            <Label htmlFor="keywords">
   596→              Mots-clés{" "}
   597→              <span className="text-muted-foreground font-normal">
   598→                (séparés par des virgules)
   599→              </span>
   600→            </Label>
   601→            <Input
   602→              id="keywords"
   603→              placeholder="Ex : intelligence artificielle, apprentissage, pédagogie"
   604→              value={form.keywords}
   605→              onChange={(e) => updateForm("keywords", e.target.value)}
   606→            />
   607→          </div>
   608→          <div className="flex flex-col gap-2">
   609→            <Label htmlFor="questions">Questions de recherche</Label>
