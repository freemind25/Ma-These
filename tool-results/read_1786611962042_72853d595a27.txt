    10→interface TypeLacune {
    11→  id: string;
    12→  label: string;
    13→  categorie: "épistémologique" | "procédurale" | "conceptuelle-contextuelle";
    14→  definitionCourte: string;
    15→  critereDiagnostic: string;
    16→  questionGuide: string;

    19→/** Un test statistique avec ses critères de sélection (RB-4) */
    20→interface TestStatistique {
    21→  id: string;
    22→  label: string;
    23→  typeDonnees: ("continue" | "catégorielle" | "ordinale")[];

    29→  conditionSpeciale?: string;
