   203→/** Construit l'insert des anti-patterns par section */
   204→function insererAntipatterns(sections?: string[]): string {
   205→  const ap = CORPUS_ECRIRE_ARTICLE_SCIENTIFIQUE.antipatterns;
   206→  const sectionsAInclure = sections ?? (Object.keys(ap) as (keyof typeof ap)[]);
   207→  const blocs = sectionsAInclure
   208→    .filter((s) => s in ap)
   209→    .map((s) => {
   210→      const section = s as keyof typeof ap;
   211→      const items = ap[section].map(
   212→        (a) => `    - ${a.erreur} → ${a.correction}`
   213→      );
   214→      return `  ${s.toUpperCase()} :
   215→${items.join("\n")}`;
   216→    });
   217→  return `ANTI-PATTERNS (ERREURS FRÉQUENTES À ÉVITER) :
   218→${blocs.join("\n\n")}`;
   219→}

   271→/** Construit l'insert des questions d'évaluation critique (Pyrczak) */
   272→function insererQuestionsEvaluationPyrczak(): string {
   273→  const categories = new Map<string, typeof QUESTIONS_EVALUATION_RECHERCHE>();
   274→  for (const q of QUESTIONS_EVALUATION_RECHERCHE) {
   275→    if (!categories.has(q.categorie)) categories.set(q.categorie, []);
   276→    categories.get(q.categorie)!.push(q);
   277→  }
   278→  const blocs = Array.from(categories.entries()).map(
   279→    ([cat, questions]) =>
   280→      `  ${cat.toUpperCase()} :
   281→${questions.map((q) => `    ${q.question}\n      → Ce chercher : ${q.ceChercher}`).join("\n")}`
   282→  );
   283→  return `GRILLE D'ÉVALUATION CRITIQUE (Pyrczak) :
   284→${blocs.join("\n\n")}`;
   285→}
