   133→function formatTime(seconds: number): string {

   459→// ─── Freewriting Exercise ──────────────────────────────────────────────
   460→
   461→function FreewritingExercise() {
   462→  const [duration, setDuration] = useState(5);
   463→  const [timeLeft, setTimeLeft] = useState(5 * 60);
   464→  const [isRunning, setIsRunning] = useState(false);
   465→  const [text, setText] = useState("");
   466→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   467→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   468→
   469→  // Timer countdown + auto-stop when time runs out
   470→  useEffect(() => {
   471→    if (isRunning && timeLeft > 0) {
   472→      intervalRef.current = setInterval(() => {
   473→        setTimeLeft((prev) => {
   474→          if (prev <= 1) {
   475→            setIsRunning(false);
   476→            if (intervalRef.current) clearInterval(intervalRef.current);
   477→            return 0;
   478→          }
   479→          return prev - 1;
   480→        });
   481→      }, 1000);
   482→    }
   483→    return () => {
   484→      if (intervalRef.current) clearInterval(intervalRef.current);
   485→    };
   486→  }, [isRunning, timeLeft]);
   487→
   488→  const handleDurationChange = (d: number) => {
   489→    setDuration(d);
   490→    if (!isRunning) setTimeLeft(d * 60);
   491→  };
   492→
   493→  const handleStart = () => {
   494→    if (!isRunning && timeLeft === duration * 60) {
   495→      setTimeLeft(duration * 60);
   496→    }
   497→    setIsRunning(true);
   498→    textareaRef.current?.focus();
   499→  };
   500→
   501→  const handlePause = () => {
   502→    setIsRunning(false);
   503→    if (intervalRef.current) clearInterval(intervalRef.current);
   504→  };
   505→
   506→  const handleReset = () => {
   507→    setIsRunning(false);
   508→    if (intervalRef.current) clearInterval(intervalRef.current);
   509→    setTimeLeft(duration * 60);
   510→    setText("");
   511→  };
   512→
   513→  const progress = ((duration * 60 - timeLeft) / (duration * 60)) * 100;
   514→  const wordCount = text.trim().split(/\s+/).filter(Boolean).length;
   515→
   516→  return (
   517→    <Card className="flex flex-col">
   518→      <CardHeader>
   519→        <CardTitle className="text-base flex items-center gap-2">
   520→          <PenTool className="h-4 w-4 text-chart-1" />
   521→          Écriture libre chronométrée
   522→        </CardTitle>
   523→        <CardDescription>
   524→          Écrivez sans vous arrêter, sans juger. Laissez couler.
   525→        </CardDescription>
   526→      </CardHeader>
   527→      <CardContent className="flex flex-col gap-4 flex-1">
   528→        {/* Duration selector */}
   529→        <div className="flex gap-2">
   530→          {FREEWRITING_DURATIONS.map((d) => (
   531→            <Button
   532→              key={d.value}
   533→              variant={duration === d.value ? "default" : "outline"}
   534→              size="sm"
   535→              onClick={() => handleDurationChange(d.value)}
   536→              disabled={isRunning}
   537→              className="flex-1"
   538→            >
   539→              {d.label}
   540→            </Button>
   541→          ))}
   542→        </div>
   543→
   544→        {/* Timer display */}
   545→        <div className="flex flex-col items-center gap-2">
   546→          <span className={`text-4xl font-mono font-bold tabular-nums ${timeLeft === 0 ? "text-chart-2" : ""}`}>
   547→            {formatTime(timeLeft)}
   548→          </span>
   549→          <Progress value={timeLeft === 0 ? 100 : progress} className="h-2 w-full" />
   550→          {timeLeft === 0 && (
   551→            <Badge variant="secondary" className="bg-chart-2/10 text-chart-2">
   552→              Temps écoulé !
   553→            </Badge>
   554→          )}
   555→        </div>
   556→
   557→        {/* Textarea */}
   558→        <Textarea
