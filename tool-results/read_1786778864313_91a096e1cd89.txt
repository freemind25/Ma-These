   480→
   481→function getCategoryScore(
   482→  category: CompetencyCategory,
   483→  skills: Record<string, ProficiencyLevel>
   484→): number {
   485→  const catSkills = category.skills;
   486→  if (catSkills.length === 0) return 0;
   487→  const total = catSkills.reduce(
   488→    (sum, s) => sum + (PROFICIENCY_VALUES[skills[s.id] || "Débutant"] || 0),
   489→    0
   490→  );
   491→  return Math.round(total / catSkills.length);
   492→}
   493→
   494→function getOverallScore(skills: Record<string, ProficiencyLevel>): number {
   495→  const allSkills = COMPETENCY_CATEGORIES.flatMap((c) => c.skills);
   496→  if (allSkills.length === 0) return 0;
   497→  const total = allSkills.reduce(
   498→    (sum, s) => sum + (PROFICIENCY_VALUES[skills[s.id] || "Débutant"] || 0),
   499→    0
   500→  );
   501→  return Math.round(total / allSkills.length);
   502→}
   503→
   504→function getScoreLabel(score: number): string {
   505→  if (score >= 90) return "Expert";
   506→  if (score >= 70) return "Avancé";
   507→  if (score >= 45) return "Intermédiaire";
   508→  return "Débutant";
   509→}
   510→
   511→function getScoreColor(score: number): string {
   512→  if (score >= 90) return "text-purple-600 dark:text-purple-400";
   513→  if (score >= 70) return "text-emerald-600 dark:text-emerald-400";
   514→  if (score >= 45) return "text-sky-600 dark:text-sky-400";
   515→  return "text-amber-600 dark:text-amber-400";
   516→}
