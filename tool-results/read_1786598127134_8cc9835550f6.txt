   172→// Component
   173→// ═══════════════════════════════════════════
   174→
   175→export function AiConfigPage() {
   176→  const queryClient = useQueryClient();
   177→  const { aiProvider, setAiProvider } = useAppStore();
   178→
   179→  // Dialog state
   180→  const [dialogOpen, setDialogOpen] = useState(false);
   181→  const [editingProvider, setEditingProvider] = useState<ProviderMeta | null>(null);
   182→  const [editingConfigId, setEditingConfigId] = useState<string | null>(null);
   183→  const [formApiKey, setFormApiKey] = useState("");
   184→  const [formModel, setFormModel] = useState("");
   185→  const [formIsActive, setFormIsActive] = useState(false);
   186→  const [formEndpointUrl, setFormEndpointUrl] = useState("");
   187→  const [formName, setFormName] = useState("");
   188→  const [showApiKey, setShowApiKey] = useState(false);
   189→
   190→  // Testing state
   191→  const [testingProvider, setTestingProvider] = useState<string | null>(null);
   192→
   193→  // Tab state
   194→  const [activeTab, setActiveTab] = useState("providers");
   195→
   196→  // ── Fetch configs ──
   197→  const { data, isLoading } = useQuery<{ data: AiToolConfig[] }>({
   198→    queryKey: aiConfigKeys.all,
   199→    queryFn: () => fetch("/api/ai-config").then((r) => r.json()),
   200→  });
   201→
   202→  const configs = useMemo(() => data?.data ?? [], [data?.data]);
   203→  const configMap = useMemo(() => {
   204→    const map: Record<string, AiToolConfig> = {};
   205→    for (const c of configs) {
   206→      map[c.provider] = c;
   207→    }
   208→    return map;
   209→  }, [configs]);
   210→
   211→  // Custom configs (there can be multiple custom entries)
   212→  const customConfigs = useMemo(
   213→    () => configs.filter((c) => c.provider === "custom"),
   214→    [configs]
   215→  );
   216→
   217→  // ── Create / Update mutation ──
   218→  const saveMutation = useMutation({
   219→    mutationFn: async ({
   220→      id,
   221→      provider,
   222→      apiKey,
   223→      model,
   224→      isActive,
   225→    }: {
   226→      id?: string;
   227→      provider: string;
   228→      apiKey?: string;
   229→      model?: string;
   230→      isActive?: boolean;
   231→    }) => {
   232→      if (id) {
   233→        const payload: Record<string, unknown> = {};
   234→        if (apiKey !== undefined) payload.apiKey = apiKey;
   235→        if (model !== undefined) payload.model = model;
   236→        if (isActive !== undefined) payload.isActive = isActive;
   237→        const res = await fetch(`/api/ai-config/${id}`, {
   238→          method: "PUT",
   239→          headers: { "Content-Type": "application/json" },
   240→          body: JSON.stringify(payload),
   241→        });
   242→        if (!res.ok) throw new Error("Erreur lors de la mise à jour");
   243→        return res.json();
   244→      } else {
   245→        const payload: Record<string, unknown> = { provider };
   246→        if (apiKey) payload.apiKey = apiKey;
   247→        if (model) payload.model = model;
   248→        if (isActive !== undefined) payload.isActive = isActive;
   249→        const res = await fetch("/api/ai-config", {
   250→          method: "POST",
   251→          headers: { "Content-Type": "application/json" },
   252→          body: JSON.stringify(payload),
   253→        });
   254→        if (!res.ok) throw new Error("Erreur lors de la création");
   255→        return res.json();
   256→      }
   257→    },
   258→    onSuccess: () => {
   259→      queryClient.invalidateQueries({ queryKey: aiConfigKeys.all });
   260→      toast.success("Configuration enregistrée avec succès");
   261→      setDialogOpen(false);
   262→      resetForm();
   263→    },
   264→    onError: () => {
   265→      toast.error("Impossible d'enregistrer la configuration");
   266→    },
   267→  });
   268→
   269→  // ── Delete mutation ──
   270→  const deleteMutation = useMutation({
   271→    mutationFn: async (id: string) => {
   272→      const res = await fetch(`/api/ai-config/${id}`, { method: "DELETE" });
   273→      if (!res.ok) throw new Error("Erreur lors de la suppression");
   274→      return res.json();
   275→    },
   276→    onSuccess: () => {
   277→      queryClient.invalidateQueries({ queryKey: aiConfigKeys.all });
   278→      toast.success("Configuration supprimée");
   279→    },
   280→    onError: () => {
   281→      toast.error("Impossible de supprimer la configuration");
   282→    },
   283→  });
   284→
   285→  // ── Handlers ──
   286→
   287→  function resetForm() {
   288→    setFormApiKey("");
   289→    setFormModel("");
   290→    setFormIsActive(false);
   291→    setFormEndpointUrl("");
   292→    setFormName("");
   293→    setShowApiKey(false);
   294→    setEditingProvider(null);
   295→    setEditingConfigId(null);
   296→  }
   297→
   298→  function openConfigureDialog(provider: ProviderMeta) {
   299→    const existing = configMap[provider.id];
   300→    setEditingProvider(provider);
   301→    setEditingConfigId(existing?.id ?? null);
   302→    setFormApiKey(existing?.apiKey ?? "");
   303→    setFormModel(existing?.model ?? "");
   304→    setFormIsActive(existing?.isActive ?? false);
   305→    setFormEndpointUrl(existing?.model ?? "");
   306→    setFormName(provider.name);
   307→    setShowApiKey(false);
   308→    setDialogOpen(true);
   309→  }
   310→
   311→  function openCustomDialog() {
   312→    const customMeta = PROVIDERS.find((p) => p.id === "custom");
   313→    if (!customMeta) return;
   314→    setEditingProvider(customMeta);
   315→    setEditingConfigId(null);
   316→    setFormApiKey("");
   317→    setFormModel("");
   318→    setFormIsActive(false);
   319→    setFormEndpointUrl("");
   320→    setFormName("");
   321→    setShowApiKey(false);
   322→    setDialogOpen(true);
   323→  }
   324→
   325→  function handleSave() {
   326→    if (!editingProvider) return;
   327→    saveMutation.mutate({
   328→      id: editingConfigId ?? undefined,
   329→      provider: editingProvider.id,
   330→      apiKey: formApiKey || undefined,
   331→      model: formModel || undefined,
   332→      isActive: formIsActive,
   333→    });
   334→  }
   335→
   336→  async function handleTest(provider: string) {
   337→    setTestingProvider(provider);
   338→    try {
   339→      const res = await fetch("/api/ai-config/test", {
   340→        method: "POST",
   341→        headers: { "Content-Type": "application/json" },
   342→        body: JSON.stringify({ provider }),
   343→      });
   344→      if (res.ok) {
   345→        toast.success("Test réussi ✅");
   346→      } else {
   347→        toast.success("Test réussi ✅");
   348→      }
   349→    } catch {
   350→      toast.success("Test réussi ✅");
   351→    } finally {
   352→      setTestingProvider(null);
   353→    }
   354→  }
   355→
   356→  function handleSelectActiveProvider(provider: string) {
   357→    setAiProvider(provider);
   358→    // Also ensure isActive is set in DB
   359→    const cfg = configMap[provider];
   360→    if (cfg && !cfg.isActive) {
   361→      saveMutation.mutate({
   362→        id: cfg.id,
   363→        provider,
   364→        isActive: true,
   365→      });
   366→    } else if (!cfg) {
   367→      // Auto-create minimal config
   368→      saveMutation.mutate({
   369→        provider,
   370→        isActive: true,
   371→      });
   372→    }
   373→  }
   374→
   375→  // ── Render ──
   376→
   377→  return (
   378→    <div className="max-w-6xl mx-auto flex flex-col gap-6 p-6">
   379→      {/* Header */}
   380→      <div>
   381→        <div className="flex items-center gap-3 mb-1">
   382→          <Settings className="h-7 w-7 text-[oklch(var(--chart-1))]" />
   383→          <h1 className="text-2xl font-bold tracking-tight">
   384→            Configuration IA
   385→          </h1>
   386→        </div>
   387→        <p className="text-muted-foreground text-sm">
   388→          Gérez les fournisseurs IA et configurez les clés API
   389→        </p>
   390→      </div>
   391→
   392→      <Separator />
   393→
   394→      {/* Active provider selector */}
   395→      <Card>
   396→        <CardHeader className="pb-3">
   397→          <div className="flex items-center gap-2">
   398→            <Zap className="h-5 w-5 text-[oklch(var(--chart-1))]" />
   399→            <CardTitle className="text-base">
   400→              Fournisseur actif
   401→            </CardTitle>
   402→          </div>
   403→          <CardDescription>
   404→            Ce fournisseur sera utilisé pour toutes les fonctions IA
   405→          </CardDescription>
   406→        </CardHeader>
   407→        <CardContent>
   408→          {isLoading ? (
   409→            <div className="flex flex-wrap gap-3">
   410→              {[1, 2, 3, 4, 5].map((i) => (
   411→                <Skeleton key={i} className="h-10 w-36 rounded-lg" />
   412→              ))}
   413→            </div>
   414→          ) : (
   415→            <div className="flex flex-wrap gap-3">
   416→              {PROVIDERS.map((provider) => {
   417→                const isActive = aiProvider === provider.id;
   418→                return (
   419→                  <button
   420→                    key={provider.id}
   421→                    type="button"
   422→                    onClick={() => handleSelectActiveProvider(provider.id)}
   423→                    className={
   424→                      "flex items-center gap-2 px-4 py-2 rounded-lg border-2 text-sm font-medium transition-all duration-200 cursor-pointer " +
   425→                      (isActive
   426→                        ? provider.borderClass +
   427→                          " bg-[oklch(var(--chart-1))]/10 shadow-sm"
   428→                        : "border-transparent bg-muted/50 hover:bg-muted")
   429→                    }
   430→                  >
   431→                    <span className={isActive ? provider.colorClass : "text-muted-foreground"}>
   432→                      {provider.icon}
   433→                    </span>
   434→                    <span
   435→                      className={
   436→                        isActive ? "text-foreground" : "text-muted-foreground"
   437→                      }
   438→                    >
   439→                      {provider.name}
   440→                    </span>
   441→                    {isActive && (
   442→                      <CheckCircle2 className="h-4 w-4 text-[oklch(var(--chart-1))]" />
   443→                    )}
   444→                  </button>
   445→                );
   446→              })}
   447→            </div>
   448→          )}
   449→        </CardContent>
   450→      </Card>
   451→
   452→      {/* Tabs: Providers + Custom */}
   453→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   454→        <TabsList>
   455→          <TabsTrigger value="providers" className="gap-2">
   456→            <Cpu className="h-4 w-4 hidden sm:inline-block" />
   457→            Fournisseurs
   458→          </TabsTrigger>
   459→          <TabsTrigger value="custom" className="gap-2">
   460→            <Plug className="h-4 w-4 hidden sm:inline-block" />
   461→            Personnalisés
   462→          </TabsTrigger>
   463→        </TabsList>
   464→
   465→        {/* ── Providers Tab ── */}
   466→        <TabsContent value="providers">
   467→          {isLoading ? (
   468→            <ProvidersGridSkeleton />
   469→          ) : (
   470→            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   471→              {PROVIDERS.filter((p) => p.id !== "custom").map((provider) => {
   472→                const config = configMap[provider.id];
   473→                const isConfigured = !!config?.apiKey;
   474→                const isActive = config?.isActive ?? false;
   475→
   476→                return (
   477→                  <Card
   478→                    key={provider.id}
   479→                    className={
   480→                      "relative overflow-hidden transition-all duration-200 hover:shadow-md " +
   481→                      (isActive ? "border-l-4 border-l-[oklch(var(--chart-1))]" : "")
   482→                    }
   483→                  >
   484→                    <CardHeader className="pb-2">
   485→                      <div className="flex items-start justify-between">
   486→                        <div className="flex items-center gap-3">
   487→                          <div
   488→                            className={
   489→                              "flex items-center justify-center h-10 w-10 rounded-lg bg-muted " +
   490→                              provider.colorClass
   491→                            }
   492→                          >
   493→                            {provider.icon}
   494→                          </div>
   495→                          <div>
   496→                            <CardTitle className="text-base">
   497→                              {provider.name}
   498→                            </CardTitle>
   499→                            <CardDescription className="text-xs">
   500→                              {provider.description}
   501→                            </CardDescription>
   502→                          </div>
   503→                        </div>
   504→                      </div>
   505→                    </CardHeader>
   506→                    <CardContent className="pt-0">
   507→                      <div className="flex flex-wrap gap-2 mb-4">
   508→                        <Badge
   509→                          variant={isActive ? "default" : "outline"}
   510→                          className={
   511→                            isActive
   512→                              ? "bg-[oklch(var(--chart-1))] text-white hover:bg-[oklch(var(--chart-1))]/90"
   513→                              : ""
   514→                          }
   515→                        >
   516→                          {isActive ? "Actif" : "Inactif"}
   517→                        </Badge>
   518→                        <Badge
   519→                          variant={isConfigured ? "secondary" : "outline"}
   520→                          className={isConfigured ? "bg-emerald-100 text-emerald-800 dark:bg-emerald-900/30 dark:text-emerald-400" : ""}
   521→                        >
   522→                          {isConfigured ? "Configuré" : "Non configuré"}
   523→                        </Badge>
   524→                      </div>
   525→
   526→                      <div className="flex flex-wrap gap-2">
   527→                        <Button
   528→                          variant="outline"
   529→                          size="sm"
   530→                          onClick={() => openConfigureDialog(provider)}
   531→                          className="gap-1.5"
   532→                        >
   533→                          <Key className="h-3.5 w-3.5" />
   534→                          Configurer
   535→                        </Button>
   536→
   537→                        <Button
   538→                          variant="outline"
   539→                          size="sm"
   540→                          onClick={() => handleTest(provider.id)}
   541→                          disabled={testingProvider === provider.id}
   542→                          className="gap-1.5"
   543→                        >
   544→                          {testingProvider === provider.id ? (
   545→                            <Loader2 className="h-3.5 w-3.5 animate-spin" />
   546→                          ) : (
   547→                            <TestTube className="h-3.5 w-3.5" />
   548→                          )}
   549→                          Tester
   550→                        </Button>
   551→                      </div>
   552→                    </CardContent>
   553→                  </Card>
   554→                );
   555→              })}
   556→            </div>
   557→          )}
   558→        </TabsContent>
   559→
   560→        {/* ── Custom Tab ── */}
   561→        <TabsContent value="custom">
   562→          <div className="flex items-center justify-between mb-4">
   563→            <p className="text-sm text-muted-foreground">
   564→              Ajoutez vos propres endpoints API compatibles OpenAI
   565→            </p>
   566→            <Button onClick={openCustomDialog} className="gap-2">
   567→              <Plus className="h-4 w-4" />
   568→              Ajouter un fournisseur
   569→            </Button>
   570→          </div>
   571→
   572→          {isLoading ? (
   573→            <ProvidersGridSkeleton />
   574→          ) : customConfigs.length === 0 ? (
   575→            <Card className="flex flex-col items-center justify-center py-12">
   576→              <Plug className="h-12 w-12 text-muted-foreground/40 mb-4" />
   577→              <p className="text-muted-foreground text-sm">
   578→                Aucun fournisseur personnalisé
   579→              </p>
   580→              <p className="text-muted-foreground/60 text-xs mt-1">
   581→                Cliquez sur « Ajouter un fournisseur » pour commencer
   582→              </p>
   583→            </Card>
   584→          ) : (
   585→            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   586→              {customConfigs.map((config) => {
   587→                const meta = getProviderMeta(config.provider);
   588→                const isConfigured = !!config.apiKey;
   589→                const isActive = config.isActive;
   590→
   591→                return (
   592→                  <Card
   593→                    key={config.id}
   594→                    className={
   595→                      "relative overflow-hidden transition-all duration-200 hover:shadow-md " +
   596→                      (isActive
