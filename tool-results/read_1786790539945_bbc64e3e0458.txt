   178→
   179→  // ─── Fetch Sources ───
   180→  const { data: sources, isLoading: sourcesLoading } = useQuery({
   181→    queryKey: ["sources"],
   182→    queryFn: async () => {
   183→      const res = await fetch("/api/sources");
   184→      const json = await res.json();
   185→      return json.data as ResearchSource[];
   186→    },
   187→  });
   188→
   189→  // ─── Fetch Entries ───
   190→  const { data: entries, isLoading: entriesLoading } = useQuery({
   191→    queryKey: ["entries"],
   192→    queryFn: async () => {
   193→      const res = await fetch("/api/entries");

   197→  });
   198→
   199→  // ─── Create Source ───
   200→  const createSource = useMutation({
   201→    mutationFn: async (data: {
   202→      title: string;
   203→      authors: string;
   204→      year: string;
   205→      type: string;
   206→      url: string;
   207→      notes: string;
   208→    }) => {
   209→      const payload: Record<string, unknown> = {
   210→        title: data.title,
   211→        type: data.type,
   212→      };
   213→      if (data.authors.trim()) payload.authors = data.authors.trim();
   214→      if (data.year) payload.year = parseInt(data.year, 10);
   215→      if (data.url.trim()) payload.url = data.url.trim();
   216→      if (data.notes.trim()) payload.notes = data.notes.trim();
   217→
   218→      const res = await fetch("/api/sources", {
   219→        method: "POST",
   220→        headers: { "Content-Type": "application/json" },
   221→        body: JSON.stringify(payload),
   222→      });
   223→      if (!res.ok) {
   224→        const err = await res.json();
   225→        throw new Error(err.error || "Erreur lors de la création");
   226→      }
   227→      return res.json();
   228→    },
   229→    onSuccess: () => {
   230→      queryClient.invalidateQueries({ queryKey: ["sources"] });
   231→      setAddSourceOpen(false);
   232→      toast.success("Source ajoutée avec succès");
   233→    },
   234→    onError: (err) => {
   235→      toast.error(err.message);
   236→    },
   237→  });
   238→
   239→  // ─── Create Entry ───
   240→  const createEntry = useMutation({
   241→    mutationFn: async (data: {
   242→      question: string;
   243→      answer: string;
   244→      tags: string;
   245→      sourceId: string;

   248→        question: data.question,
   249→        answer: data.answer,
   250→      };

   252→      if (data.sourceId) payload.sourceId = data.sourceId;
   253→
   254→      const res = await fetch("/api/entries", {
   255→        method: "POST",
   256→        headers: { "Content-Type": "application/json" },
   257→        body: JSON.stringify(payload),
   258→      });

   260→        const err = await res.json();
   261→        throw new Error(err.error || "Erreur lors de la création");
   262→      }
   263→      return res.json();
   264→    },
   265→    onSuccess: () => {
   266→      queryClient.invalidateQueries({ queryKey: ["entries"] });
   267→      queryClient.invalidateQueries({ queryKey: ["sources"] });
   268→      setAddEntryOpen(false);
   269→      toast.success("Entrée ajoutée avec succès");

   272→      toast.error(err.message);
   273→    },

   275→
   276→  // ─── Delete Source ───

   278→    mutationFn: async (id: string) => {
   279→      await fetch(`/api/sources/${id}`, { method: "DELETE" });

   282→      queryClient.invalidateQueries({ queryKey: ["sources"] });

   284→    },
   285→  });

   718→function AiConsensusTab() {
   719→  const [prompt, setPrompt] = useState("");
   720→  const [selectedModes, setSelectedModes] = useState<Set<string>>(
   721→    new Set(["scientific-writing", "peer-review"])
   722→  );
   723→  const [results, setResults] = useState<
   724→    Array<{ mode: string; content: string }>
   725→  >([]);
   726→  const [isComparing, setIsComparing] = useState(false);
   727→
   728→  const toggleMode = (modeId: string) => {
   729→    setSelectedModes((prev) => {
   730→      const next = new Set(prev);
   731→      if (next.has(modeId)) {
   732→        next.delete(modeId);
   733→      } else {
   734→        next.add(modeId);
   735→      }
   736→      return next;
   737→    });
   738→  };
   739→
   740→  const handleCompare = async () => {
   741→    if (!prompt.trim() || selectedModes.size === 0) return;
   742→
   743→    setIsComparing(true);
   744→    setResults([]);
   745→    const newResults: Array<{ mode: string; content: string }> = [];
   746→
   747→    for (const modeId of selectedModes) {
   748→      try {
   749→        const res = await fetch("/api/ai-writing", {
   750→          method: "POST",
   751→          headers: { "Content-Type": "application/json" },
   752→          body: JSON.stringify({ mode: modeId, prompt: prompt.trim() }),
   753→        });
   754→        if (!res.ok) {
   755→          const err = await res.json();
   756→          newResults.push({
   757→            mode: modeId,
   758→            content: `Erreur : ${err.error || "Erreur de génération"}`,
   759→          });
   760→        } else {
   761→          const json = await res.json();
   762→          newResults.push({
   763→            mode: modeId,
   764→            content: json.data.content,
   765→          });
   766→        }
   767→      } catch {
   768→        newResults.push({
   769→          mode: modeId,
   770→          content: "Erreur de connexion au service IA",
   771→        });
   772→      }
   773→    }
   774→
   775→    setResults(newResults);
   776→    setIsComparing(false);
   777→  };
   778→
   779→  const getModeInfo = (modeId: string) =>
   780→    AI_CONSENSUS_MODES.find((m) => m.id === modeId);
   781→
   782→  return (
   783→    <div className="flex flex-col gap-6">
   784→      {/* Input Section */}
   785→      <Card>
   786→        <CardHeader className="pb-3">
   787→          <CardTitle className="text-sm flex items-center gap-2">
   788→            <MessageSquare className="h-4 w-4" />
   789→            Comparaison multi-modes
   790→          </CardTitle>
   791→          <CardDescription className="text-xs">
   792→            Saisissez un texte ou une question et comparez les réponses de plusieurs modes IA
   793→          </CardDescription>
   794→        </CardHeader>
   795→        <CardContent className="flex flex-col gap-4">
   796→          <Textarea
   797→            placeholder="Saisissez votre texte, question ou extrait à analyser..."
   798→            value={prompt}
   799→            onChange={(e) => setPrompt(e.target.value)}
   800→            rows={5}
   801→            className="resize-none text-sm"
   802→            disabled={isComparing}
   803→          />
   804→
   805→          {/* Mode Selection */}
   806→          <div className="flex flex-col gap-2">
   807→            <label className="text-sm font-medium">Modes IA à comparer</label>
   808→            <div className="flex flex-wrap gap-2">
   809→              {AI_CONSENSUS_MODES.map((mode) => {
   810→                const Icon = mode.icon;
   811→                const isSelected = selectedModes.has(mode.id);
   812→                return (
   813→                  <button
   814→                    key={mode.id}
   815→                    onClick={() => toggleMode(mode.id)}
   816→                    className={`flex items-center gap-2 rounded-lg border px-3 py-2 text-sm transition-all hover:bg-muted/50 ${
   817→                      isSelected

   821→                  >
   822→                    <Icon className="h-4 w-4" />
   823→                    {mode.label}
   824→                  </button>
   825→                );
   826→              })}
   827→            </div>
   828→          </div>
   829→
   830→          <div className="flex items-center justify-between">
   831→            <span className="text-xs text-muted-foreground">

   833→            </span>
   834→            <Button
   835→              onClick={handleCompare}
   836→              disabled={!prompt.trim() || selectedModes.size === 0 || isComparing}

   855→      {/* Results */}

   908→          {/* Synthèse */}

   911→              <CardHeader className="pb-3">
