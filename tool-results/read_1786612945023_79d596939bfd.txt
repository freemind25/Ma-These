   180→/** Construit l'insert des anti-patterns par section */
   181→function insererAntipatterns(sections?: string[]): string {
   182→  const ap = CORPUS_ECRIRE_ARTICLE_SCIENTIFIQUE.antipatterns;
   183→  const sectionsAInclure = sections ?? (Object.keys(ap) as (keyof typeof ap)[]);
   184→  const blocs = sectionsAInclure
   185→    .filter((s) => s in ap)
   186→    .map((s) => {
   187→      const section = s as keyof typeof ap;
   188→      const items = ap[section].map(
   189→        (a) => `    - ${a.erreur} → ${a.correction}`
   190→      );
   191→      return `  ${s.toUpperCase()} :
   192→${items.join("\n")}`;
   193→    });
   194→  return `ANTI-PATTERNS (ERREURS FRÉQUENTES À ÉVITER) :
   195→${blocs.join("\n\n")}`;
   196→}
