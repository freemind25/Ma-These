  1073→// SUB-COMPONENT — Pomodoro Section
  1074→// ═══════════════════════════════════════════════════════════════════════
  1075→
  1076→function PomodoroSection() {
  1077→  const WORK_DURATION = 25;
  1078→  const BREAK_DURATION = 5;
  1079→
  1080→  const [timeLeft, setTimeLeft] = useState(WORK_DURATION * 60);
  1081→  const [isRunning, setIsRunning] = useState(false);
  1082→  const [isBreak, setIsBreak] = useState(false);
  1083→  const [completedSessions, setCompletedSessions] = useState(0);
  1084→  const [totalWorkTime, setTotalWorkTime] = useState(0);
  1085→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
  1086→
  1087→  const currentDuration = isBreak ? BREAK_DURATION : WORK_DURATION;
  1088→
  1089→  useEffect(() => {
  1090→    if (isRunning && timeLeft > 0) {
  1091→      intervalRef.current = setInterval(() => {
  1092→        setTimeLeft((prev) => {
  1093→          if (prev <= 1) {
  1094→            setIsRunning(false);
  1095→            if (intervalRef.current) clearInterval(intervalRef.current);
  1096→            if (!isBreak) {
  1097→              setCompletedSessions((c) => c + 1);
  1098→              setIsBreak(true);
  1099→              return BREAK_DURATION * 60;
  1100→            } else {
  1101→              setIsBreak(false);
  1102→              return WORK_DURATION * 60;
  1103→            }
  1104→          }
  1105→          if (!isBreak) {
  1106→            setTotalWorkTime((t) => t + 1);
  1107→          }
  1108→          return prev - 1;
  1109→        });
  1110→      }, 1000);
  1111→    }
  1112→    return () => {
  1113→      if (intervalRef.current) clearInterval(intervalRef.current);
  1114→    };
  1115→  }, [isRunning, timeLeft, isBreak]);
  1116→
  1117→  const handleStartPause = () => {
  1118→    if (isRunning) {
  1119→      setIsRunning(false);
  1120→      if (intervalRef.current) clearInterval(intervalRef.current);
  1121→    } else {
  1122→      setIsRunning(true);
  1123→    }
  1124→  };
  1125→
  1126→  const handleReset = () => {
  1127→    setIsRunning(false);
  1128→    setIsBreak(false);
  1129→    if (intervalRef.current) clearInterval(intervalRef.current);
  1130→    setTimeLeft(WORK_DURATION * 60);
