    10→/** Un principe fondamental d'ingénierie de prompts */
    11→interface PrincipePrompt {
    12→  id: string;
    13→  titre: string;
    14→  categorie: "structure" | "clarte" | "specificite" | "contraintes" | "iteration" | "contexte";
    15→  description: string;
    16→  exempleAvant: string;
    17→  exempleApres: string;
    18→}
    19→
    20→/** Un modèle de prompt de recherche spécialisé */
    21→interface ModelePromptRecherche {
    22→  id: string;
    23→  titre: string;
    24→  contexte: string;
    25→  structure: string[];
    26→  variables: { nom: string; description: string; obligatoire: boolean }[];
    27→  conseilExecution: string;
    28→}
    29→
    30→/** Un anti-pattern de prompting avec sa correction */
    31→interface AntiPatternPrompt {
    32→  id: string;
    33→  erreur: string;
    34→  consequence: string;
    35→  correction: string;
    36→  categorie: "vague" | "surcharge" | "absence_contexte" | "mauvais_role" | "absence_contraintes" | "iteration_unique";
    37→}
    38→
    39→/** Une règle empirique de rédaction d'articles de recherche */
    40→interface RegleEmpirique {
    41→  id: string;
    42→  domaine: "problematique" | "methodologie" | "resultats" | "discussion" | "redaction" | "structure" | "submission";
    43→  regle: string;
    44→  justification: string;
    45→}
