   175→function ResearchNotebookTab() {
   176→  const queryClient = useQueryClient();
   177→  const [addSourceOpen, setAddSourceOpen] = useState(false);
   178→  const [addEntryOpen, setAddEntryOpen] = useState(false);
   179→
   180→  // ─── Fetch Sources ───
   181→  const { data: sources, isLoading: sourcesLoading } = useQuery({
   182→    queryKey: ["sources"],
   183→    queryFn: async () => {
   184→      const res = await fetch("/api/sources");
   185→      const json = await res.json();
   186→      return json.data as ResearchSource[];
   187→    },
   188→  });
   189→
   190→  // ─── Fetch Entries ───
   191→  const { data: entries, isLoading: entriesLoading } = useQuery({
   192→    queryKey: ["entries"],
   193→    queryFn: async () => {
   194→      const res = await fetch("/api/entries");
   195→      const json = await res.json();
   196→      return json.data as NotebookEntry[];
   197→    },
   198→  });
   199→
   200→  // ─── Create Source ───
   201→  const createSource = useMutation({
   202→    mutationFn: async (data: {
   203→      title: string;
   204→      authors: string;
   205→      year: string;
   206→      type: string;
   207→      url: string;
   208→      notes: string;
   209→    }) => {
   210→      const payload: Record<string, unknown> = {
   211→        title: data.title,
   212→        type: data.type,
   213→      };
   214→      if (data.authors.trim()) payload.authors = data.authors.trim();
   215→      if (data.year) payload.year = parseInt(data.year, 10);
   216→      if (data.url.trim()) payload.url = data.url.trim();
   217→      if (data.notes.trim()) payload.notes = data.notes.trim();
   218→
   219→      const res = await fetch("/api/sources", {
   220→        method: "POST",
   221→        headers: { "Content-Type": "application/json" },
   222→        body: JSON.stringify(payload),
   223→      });
   224→      if (!res.ok) {
   225→        const err = await res.json();
   226→        throw new Error(err.error || "Erreur lors de la création");
   227→      }
   228→      return res.json();
   229→    },
   230→    onSuccess: () => {
   231→      queryClient.invalidateQueries({ queryKey: ["sources"] });
   232→      setAddSourceOpen(false);
   233→      toast.success("Source ajoutée avec succès");
   234→    },
   235→    onError: (err) => {
   236→      toast.error(err.message);
   237→    },
   238→  });
   239→
   240→  // ─── Create Entry ───
   241→  const createEntry = useMutation({
   242→    mutationFn: async (data: {
   243→      question: string;
   244→      answer: string;
   245→      tags: string;
   246→      sourceId: string;
   247→    }) => {
   248→      const payload: Record<string, unknown> = {
   249→        question: data.question,
   250→        answer: data.answer,
   251→      };
   252→      if (data.tags.trim()) payload.tags = data.tags.trim();
   253→      if (data.sourceId) payload.sourceId = data.sourceId;
   254→
   255→      const res = await fetch("/api/entries", {
   256→        method: "POST",
   257→        headers: { "Content-Type": "application/json" },
   258→        body: JSON.stringify(payload),
   259→      });
   260→      if (!res.ok) {
   261→        const err = await res.json();
   262→        throw new Error(err.error || "Erreur lors de la création");
   263→      }
   264→      return res.json();
   265→    },
   266→    onSuccess: () => {
   267→      queryClient.invalidateQueries({ queryKey: ["entries"] });
   268→      queryClient.invalidateQueries({ queryKey: ["sources"] });
   269→      setAddEntryOpen(false);
   270→      toast.success("Entrée ajoutée avec succès");
   271→    },
   272→    onError: (err) => {
   273→      toast.error(err.message);
   274→    },
   275→  });
   276→
   277→  // ─── Delete Source ───
   278→  const deleteSource = useMutation({
   279→    mutationFn: async (id: string) => {
   280→      await fetch(`/api/sources/${id}`, { method: "DELETE" });
   281→    },
   282→    onSuccess: () => {
   283→      queryClient.invalidateQueries({ queryKey: ["sources"] });
   284→      toast.success("Source supprimée");
   285→    },
   286→  });
   287→
   288→  // ─── Delete Entry ───
   289→  const deleteEntry = useMutation({
   290→    mutationFn: async (id: string) => {
   291→      await fetch(`/api/entries/${id}`, { method: "DELETE" });
   292→    },
   293→    onSuccess: () => {
   294→      queryClient.invalidateQueries({ queryKey: ["entries"] });
   295→      queryClient.invalidateQueries({ queryKey: ["sources"] });
   296→      toast.success("Entrée supprimée");
   297→    },
   298→  });
   299→
   300→  return (
   301→    <div className="flex flex-col gap-6">
   302→      {/* Sources Section */}
   303→      <div className="flex flex-col gap-4">
   304→        <div className="flex items-center justify-between">
   305→          <h2 className="text-lg font-semibold flex items-center gap-2">
   306→            <StickyNote className="h-5 w-5 text-primary" />
   307→            Sources de recherche
   308→            {sources && (
   309→              <Badge variant="secondary" className="text-[10px] h-5">
   310→                {sources.length}
   311→              </Badge>
   312→            )}
   313→          </h2>
   314→          <Dialog open={addSourceOpen} onOpenChange={setAddSourceOpen}>
   315→            <DialogTrigger asChild>
   316→              <Button className="gap-2" size="sm">
   317→                <Plus className="h-4 w-4" />
   318→                Ajouter une source
   319→              </Button>
   320→            </DialogTrigger>
   321→            <DialogContent className="sm:max-w-[500px]">
   322→              <DialogHeader>
   323→                <DialogTitle>Nouvelle source de recherche</DialogTitle>
   324→                <DialogDescription>
   325→                  Ajoutez une source à votre carnet de recherche
   326→                </DialogDescription>
   327→              </DialogHeader>
   328→              <AddSourceForm
   329→                onSubmit={(data) => createSource.mutate(data)}
   330→                isPending={createSource.isPending}
   331→              />
   332→            </DialogContent>
   333→          </Dialog>
   334→        </div>
   335→
   336→        {/* Sources Grid */}
   337→        {sourcesLoading ? (
   338→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   339→            {[1, 2, 3].map((i) => (
   340→              <Skeleton key={i} className="h-36 rounded-lg" />
   341→            ))}
   342→          </div>
   343→        ) : sources && sources.length > 0 ? (
   344→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   345→            {sources.map((source) => (
   346→              <Card key={source.id} className="flex flex-col">
   347→                <CardHeader className="pb-2">
   348→                  <div className="flex items-start justify-between gap-2">
   349→                    <div className="flex-1 min-w-0">
   350→                      <CardTitle className="text-sm leading-tight line-clamp-2">
   351→                        {source.title}
   352→                      </CardTitle>
   353→                      {source.authors && (
   354→                        <CardDescription className="text-xs mt-1 truncate">
   355→                          {source.authors}
   356→                        </CardDescription>
   357→                      )}
   358→                    </div>
   359→                    <Button
   360→                      variant="ghost"
   361→                      size="icon"
   362→                      className="h-7 w-7 shrink-0"
   363→                      onClick={() => deleteSource.mutate(source.id)}
   364→                    >
   365→                      <Trash2 className="h-3.5 w-3.5 text-muted-foreground hover:text-destructive" />
   366→                    </Button>
   367→                  </div>
   368→                </CardHeader>
   369→                <CardContent className="flex flex-col gap-2 flex-1">
   370→                  <div className="flex items-center gap-2">
   371→                    <Badge
   372→                      variant="secondary"
   373→                      className={`text-[10px] h-5 ${
   374→                        SOURCE_TYPE_COLORS[source.type] || ""
   375→                      }`}
   376→                    >
   377→                      {SOURCE_TYPES.find((t) => t.value === source.type)?.label ||
   378→                        source.type}
   379→                    </Badge>
   380→                    {source.year && (
   381→                      <span className="text-xs text-muted-foreground tabular-nums">
   382→                        {source.year}
   383→                      </span>
   384→                    )}
   385→                    {source._count && source._count.entries > 0 && (
   386→                      <Badge variant="outline" className="text-[10px] h-5">
   387→                        {source._count.entries}{" "}
   388→                        {source._count.entries > 1 ? "entrées" : "entrée"}
   389→                      </Badge>
   390→                    )}
   391→                  </div>
   392→                  {source.notes && (
   393→                    <p className="text-xs text-muted-foreground line-clamp-2 flex-1">
   394→                      {source.notes}
   395→                    </p>
   396→                  )}
   397→                </CardContent>
   398→              </Card>
   399→            ))}
   400→          </div>
   401→        ) : (
   402→          <Card>
   403→            <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   404→              <BookOpen className="h-10 w-10 text-muted-foreground/30 mb-3" />
   405→              <h3 className="text-sm font-medium">Aucune source de recherche</h3>
   406→              <p className="text-xs text-muted-foreground mt-1">
   407→                Ajoutez votre première source pour commencer votre carnet de recherche
   408→              </p>
   409→            </CardContent>
   410→          </Card>
   411→        )}
   412→      </div>
   413→
   414→      <Separator />
   415→
   416→      {/* Notebook Entries Section */}
   417→      <div className="flex flex-col gap-4">
   418→        <div className="flex items-center justify-between">
   419→          <h2 className="text-lg font-semibold flex items-center gap-2">
   420→            <FileQuestion className="h-5 w-5 text-primary" />
   421→            Notes et questions
   422→            {entries && (
   423→              <Badge variant="secondary" className="text-[10px] h-5">
   424→                {entries.length}
   425→              </Badge>
   426→            )}
   427→          </h2>
   428→          <Dialog open={addEntryOpen} onOpenChange={setAddEntryOpen}>
   429→            <DialogTrigger asChild>
   430→              <Button className="gap-2" size="sm">

   451→        {/* Entries List */}

   455→              <Skeleton key={i} className="h-24 rounded-lg" />

   528→function AddSourceForm({
   529→  onSubmit,
   530→  isPending,
   531→}: {
   532→  onSubmit: (data: {
   533→    title: string;
   534→    authors: string;
   535→    year: string;
   536→    type: string;
   537→    url: string;
   538→    notes: string;
   539→  }) => void;
   540→  isPending: boolean;

   542→  const [form, setForm] = useState({
   543→    title: "",
   544→    authors: "",
   545→    year: "",
   546→    type: "article" as string,
   547→    url: "",
   548→    notes: "",
   549→  });
   550→
   551→  const handleChange = (key: string, value: string) => {
   552→    setForm((prev) => ({ ...prev, [key]: value }));

   555→  return (
   556→    <div className="grid gap-4 py-4">
   557→      <div className="grid gap-2">

   575→          <label className="text-sm font-medium">Année</label>

   577→            placeholder="2024"
   578→            value={form.year}
   579→            onChange={(e) => handleChange("year", e.target.value)}
   580→          />
   581→        </div>
   582→        <div className="grid gap-2">
   583→          <label className="text-sm font-medium">Type *</label>
   584→          <Select value={form.type} onValueChange={(v) => handleChange("type", v)}>
   585→            <SelectTrigger>
   586→              <SelectValue />
   587→            </SelectTrigger>
   588→            <SelectContent>
   589→              {SOURCE_TYPES.map((t) => (
   590→                <SelectItem key={t.value} value={t.value}>
   591→                  {t.label}
   592→                </SelectItem>
   593→              ))}
   594→            </SelectContent>
   595→          </Select>
   596→        </div>
   597→      </div>
   598→      <div className="grid gap-2">
   599→        <label className="text-sm font-medium">URL</label>
   600→        <Input
   601→          placeholder="https://..."
   602→          value={form.url}
   603→          onChange={(e) => handleChange("url", e.target.value)}
   604→        />
   605→      </div>
   606→      <div className="grid gap-2">
   607→        <label className="text-sm font-medium">Notes</label>
   608→        <Textarea
   609→          placeholder="Notes sur cette source..."
   610→          value={form.notes}
   611→          onChange={(e) => handleChange("notes", e.target.value)}
   612→          rows={3}
   613→          className="resize-none"
   614→        />
   615→      </div>
   616→      <DialogFooter>
   617→        <Button
   618→          onClick={() => onSubmit(form)}
   619→          disabled={!form.title.trim() || isPending}
   620→        >
   621→          {isPending ? "Ajout..." : "Ajouter"}
   622→        </Button>
   623→      </DialogFooter>
   624→    </div>
   625→  );
   626→}

   719→function AiConsensusTab() {
   720→  const { withAiConfig } = useAiConfig();
   721→  const [prompt, setPrompt] = useState("");
   722→  const [selectedModes, setSelectedModes] = useState<Set<string>>(
   723→    new Set(["scientific-writing", "peer-review"])
   724→  );
   725→  const [results, setResults] = useState<
   726→    Array<{ mode: string; content: string }>
   727→  >([]);
   728→  const [isComparing, setIsComparing] = useState(false);
   729→
   730→  const toggleMode = (modeId: string) => {
   731→    setSelectedModes((prev) => {
   732→      const next = new Set(prev);
   733→      if (next.has(modeId)) {
   734→        next.delete(modeId);
   735→      } else {
   736→        next.add(modeId);
   737→      }
   738→      return next;
   739→    });
   740→  };
   741→
   742→  const handleCompare = async () => {
   743→    if (!prompt.trim() || selectedModes.size === 0) return;
   744→
   745→    setIsComparing(true);
   746→    setResults([]);
   747→    const newResults: Array<{ mode: string; content: string }> = [];
   748→
   749→    for (const modeId of selectedModes) {
   750→      try {
   751→        const res = await fetch("/api/ai-writing", {
   752→          method: "POST",
   753→          headers: { "Content-Type": "application/json" },
   754→          body: JSON.stringify(withAiConfig({ mode: modeId, prompt: prompt.trim() })),
   755→        });
   756→        if (!res.ok) {
   757→          const err = await res.json();
   758→          newResults.push({
   759→            mode: modeId,
   760→            content: `Erreur : ${err.error || "Erreur de génération"}`,
   761→          });
   762→        } else {
   763→          const json = await res.json();
   764→          newResults.push({
   765→            mode: modeId,
   766→            content: json.data.content,
   767→          });
   768→        }
   769→      } catch {
   770→        newResults.push({
   771→          mode: modeId,
   772→          content: "Erreur de connexion au service IA",
   773→        });
   774→      }
   775→    }
   776→
   777→    setResults(newResults);
   778→    setIsComparing(false);
   779→  };
   780→
   781→  const getModeInfo = (modeId: string) =>
   782→    AI_CONSENSUS_MODES.find((m) => m.id === modeId);
   783→
   784→  return (
   785→    <div className="flex flex-col gap-6">
   786→      {/* Input Section */}
   787→      <Card>
   788→        <CardHeader className="pb-3">
   789→          <CardTitle className="text-sm flex items-center gap-2">
   790→            <MessageSquare className="h-4 w-4" />
   791→            Comparaison multi-modes
   792→          </CardTitle>
   793→          <CardDescription className="text-xs">
   794→            Saisissez un texte ou une question et comparez les réponses de plusieurs modes IA
   795→          </CardDescription>
   796→        </CardHeader>
   797→        <CardContent className="flex flex-col gap-4">
   798→          <Textarea
   799→            placeholder="Saisissez votre texte, question ou extrait à analyser..."
   800→            value={prompt}
   801→            onChange={(e) => setPrompt(e.target.value)}
   802→            rows={5}
   803→            className="resize-none text-sm"
   804→            disabled={isComparing}
   805→          />
   806→
   807→          {/* Mode Selection */}
   808→          <div className="flex flex-col gap-2">
   809→            <label className="text-sm font-medium">Modes IA à comparer</label>
   810→            <div className="flex flex-wrap gap-2">
   811→              {AI_CONSENSUS_MODES.map((mode) => {
   812→                const Icon = mode.icon;
   813→                const isSelected = selectedModes.has(mode.id);
   814→                return (
   815→                  <button
   816→                    key={mode.id}
   817→                    onClick={() => toggleMode(mode.id)}
   818→                    className={`flex items-center gap-2 rounded-lg border px-3 py-2 text-sm transition-all hover:bg-muted/50 ${
   819→                      isSelected
   820→                        ? "border-primary bg-primary/5 text-primary"
   821→                        : "border-border text-muted-foreground"
   822→                    }`}
   823→                  >
   824→                    <Icon className="h-4 w-4" />
   825→                    {mode.label}
   826→                  </button>
   827→                );
   828→              })}
   829→            </div>
   830→          </div>
   831→
   832→          <div className="flex items-center justify-between">
   833→            <span className="text-xs text-muted-foreground">
   834→              {selectedModes.size} mode(s) sélectionné(s) · {prompt.length} caractères
   835→            </span>
   836→            <Button
   837→              onClick={handleCompare}
   838→              disabled={!prompt.trim() || selectedModes.size === 0 || isComparing}
   839→              className="gap-2"
   840→            >
   841→              {isComparing ? (
   842→                <>
   843→                  <Loader2 className="h-4 w-4 animate-spin" />
   844→                  Analyse en cours...
   845→                </>
   846→              ) : (
   847→                <>
   848→                  <GitCompareArrows className="h-4 w-4" />
   849→                  Comparer
   850→                </>
   851→              )}
   852→            </Button>
   853→          </div>
   854→        </CardContent>
   855→      </Card>
   856→
   857→      {/* Results */}
   858→      {isComparing && (
   859→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   860→          {Array.from(selectedModes).map((modeId) => {
   861→            const mode = getModeInfo(modeId);
   862→            return (
   863→              <Card key={modeId}>
   864→                <CardHeader className="pb-3">
   865→                  <CardTitle className="text-sm flex items-center gap-2">
   866→                    {mode && <mode.icon className="h-4 w-4" />}
   867→                    {mode?.label || modeId}
   868→                  </CardTitle>
   869→                </CardHeader>
   870→                <CardContent>
   871→                  <div className="space-y-2">
   872→                    <Skeleton className="h-4 w-3/4" />
   873→                    <Skeleton className="h-4 w-full" />
   874→                    <Skeleton className="h-4 w-5/6" />
   875→                    <Skeleton className="h-4 w-2/3" />
   876→                  </div>
   877→                </CardContent>
   878→              </Card>
   879→            );
   880→          })}
   881→        </div>
   882→      )}
   883→
   884→      {results.length > 0 && !isComparing && (
   885→        <>
   886→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   887→            {results.map((result) => {
   888→              const mode = getModeInfo(result.mode);
   889→              return (
   890→                <Card key={result.mode}>
   891→                  <CardHeader className="pb-3">
   892→                    <CardTitle className="text-sm flex items-center gap-2">
   893→                      {mode && <mode.icon className="h-4 w-4 text-primary" />}
   894→                      {mode?.label || result.mode}
   895→                    </CardTitle>

   900→                        if (line.trim() === "") return <br key={i} />;
   901→                        return <p key={i}>{line}</p>;
   902→                      })}
   903→                    </div>
   904→                  </CardContent>
   905→                </Card>
   906→              );
   907→            })}
   908→          </div>
   909→
   910→          {/* Synthèse */}
   911→          {results.length >= 2 && (
   912→            <Card className="border-primary/30 bg-primary/5">
   913→              <CardHeader className="pb-3">
   914→                <CardTitle className="text-sm flex items-center gap-2">
   915→                  <Sparkles className="h-4 w-4 text-primary" />
   916→                  Synthèse comparative
   917→                </CardTitle>
   918→                <CardDescription className="text-xs">
   919→                  Vue d&apos;ensemble des réponses des {results.length} modes IA sélectionnés
   920→                </CardDescription>
   921→              </CardHeader>
   922→              <CardContent>

   924→                  {results.map((result) => {
   925→                    const mode = getModeInfo(result.mode);
   926→                    const preview =
   927→                      result.content.length > 150
   928→                        ? result.content.substring(0, 150) + "..."
   929→                        : result.content;
   930→                    return (
