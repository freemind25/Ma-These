   168→   168→function VersionRow({ version }: { version: CadrageVersion }) {
   169→   169→  const parsedSnapshot = useMemo(() => {
   170→   170→    try {
   171→   171→      const fields = JSON.parse(version.snapshot) as CadrageField[];
   172→   172→      const filled = fields.filter((f) => f.value && f.value.trim().length > 0).length;
   173→   173→      return { count: fields.length, filled };
   174→   174→    } catch {
   175→   175→      return { count: 0, filled: 0 };
   176→   176→    }
   177→   177→  }, [version.snapshot]);
   178→   178→
   179→   179→  const date = new Date(version.createdAt);
   180→   180→  const dateStr = date.toLocaleDateString("fr-FR", {
   181→   181→    day: "numeric",
   182→   182→    month: "short",
   183→   183→    year: "numeric",
   184→   184→    hour: "2-digit",
   185→   185→    minute: "2-digit",
   186→   186→  });
   187→   187→
   188→   188→  return (
   189→   189→    <div className="flex items-center justify-between rounded-lg border p-3">
   190→   190→      <div className="flex flex-col gap-1 min-w-0">
   191→   191→        <p className="text-sm font-medium truncate">
   192→   192→          {version.label ?? "Version sans titre"}
   193→   193→        </p>
   194→   194→        <p className="text-xs text-muted-foreground">{dateStr}</p>
   195→   195→      </div>
   196→   196→      <div className="flex items-center gap-2 shrink-0">
   197→   197→        <Badge variant="outline" className="text-xs tabular-nums">
   198→   198→          {parsedSnapshot.filled}/{parsedSnapshot.count} champs
   199→   199→        </Badge>
   200→   200→      </div>
   201→   201→    </div>
   202→   202→  );
   203→   203→}
