   194→/** Construit l'insert des anti-patterns par section */
   195→function insererAntipatterns(sections?: string[]): string {
   196→  const ap = CORPUS_ECRIRE_ARTICLE_SCIENTIFIQUE.antipatterns;
   197→  const sectionsAInclure = sections ?? (Object.keys(ap) as (keyof typeof ap)[]);
   198→  const blocs = sectionsAInclure
   199→    .filter((s) => s in ap)
   200→    .map((s) => {
   201→      const section = s as keyof typeof ap;
   202→      const items = ap[section].map(
   203→        (a) => `    - ${a.erreur} → ${a.correction}`
   204→      );
   205→      return `  ${s.toUpperCase()} :
   206→${items.join("\n")}`;
   207→    });
   208→  return `ANTI-PATTERNS (ERREURS FRÉQUENTES À ÉVITER) :
   209→${blocs.join("\n\n")}`;
   210→}

   262→/** Construit l'insert des questions d'évaluation critique (Pyrczak) */
   263→function insererQuestionsEvaluationPyrczak(): string {
   264→  const categories = new Map<string, typeof QUESTIONS_EVALUATION_RECHERCHE>();
   265→  for (const q of QUESTIONS_EVALUATION_RECHERCHE) {
   266→    if (!categories.has(q.categorie)) categories.set(q.categorie, []);
   267→    categories.get(q.categorie)!.push(q);
   268→  }
   269→  const blocs = Array.from(categories.entries()).map(
   270→    ([cat, questions]) =>
   271→      `  ${cat.toUpperCase()} :
   272→${questions.map((q) => `    ${q.question}\n      → Ce chercher : ${q.ceChercher}`).join("\n")}`
   273→  );
   274→  return `GRILLE D'ÉVALUATION CRITIQUE (Pyrczak) :
   275→${blocs.join("\n\n")}`;
   276→}
